"""Summarize closed run27 evidence; no ROS, bag decoding, truth, or control.

Run only after the experiment supervisor is terminal and its owned cleanup has
been confirmed. Existing reports are never overwritten. Failed evidence stays
failed; this summary is not a full visual-search or physical-robot acceptance.
"""
from collections import Counter
import hashlib
import json
import math
from pathlib import Path

ROOT = Path(__file__).resolve().parents[2]
RUN = ROOT / 'outputs/ros_smoke/run27-writer-pipe'
OUT = ROOT / 'outputs/setup/run27-outcome-review'
SOURCES, MISSING = {}, []


def load(name, required=False):
    path = RUN / name
    if not path.is_file():
        if required:
            raise RuntimeError('Missing required evidence: ' + str(path))
        MISSING.append(str(name))
        return {}
    if path.is_symlink() or path.stat().st_size > 256 * 1024 * 1024:
        raise RuntimeError('Unsafe or oversized evidence: ' + str(path))
    raw = path.read_bytes()
    SOURCES[str(path.relative_to(ROOT))] = {
        'bytes': len(raw), 'sha256': hashlib.sha256(raw).hexdigest()}
    return json.loads(raw)


def delta(a, b):
    return math.atan2(math.sin(a - b), math.cos(a - b))


def navigation_extrapolation():
    path = RUN / 'navigation.log'
    if not path.is_file():
        MISSING.append('navigation.log')
        return {'available': False, 'matched_line_count': None}
    if path.is_symlink() or path.stat().st_size > 256 * 1024 * 1024:
        raise RuntimeError('Unsafe or oversized navigation log')
    raw = path.read_bytes()
    SOURCES[str(path.relative_to(ROOT))] = {'bytes': len(raw), 'sha256': hashlib.sha256(raw).hexdigest()}
    matches = [(number, line) for number, line in enumerate(raw.decode('utf-8', errors='replace').splitlines(), 1)
               if 'extrapolat' in line.casefold()]
    return {'available': True, 'match_rule': 'case-insensitive substring extrapolat; one count per matching log line',
        'matched_line_count': len(matches),
        'future_line_count': sum('future' in line.casefold() for _, line in matches),
        'past_line_count': sum('past' in line.casefold() for _, line in matches),
        'neither_future_nor_past_line_count': sum(not any(word in line.casefold() for word in ('future', 'past')) for _, line in matches),
        'examples_first_20': [{'line_number': number, 'text': line[:2000], 'text_truncated': len(line) > 2000}
                              for number, line in matches[:20]],
        'scope': 'Log-line counts, not unique incidents or controller ticks; full original log archived. Missing lines or logs are not proof of no TF errors.'}


def accepted_file_inventory(consumer, producer_by_id):
    """Only file presence for exact accepted IDs; not a replacement strict audit."""
    expected = ('rgb.bin', 'depth.bin', 'producer.json', 'camera_info.json', 'request.json', 'manifest.json')
    root = RUN / 'retained'
    if root.is_symlink() or not root.is_dir():
        raise RuntimeError('Unsafe or missing retained root')
    frames, seen = [], set()
    for window in consumer.get('windows', []):
        for entry in window.get('accepted', []):
            rid, fingerprint = entry['request']['retention_id'], entry['request_sha256']
            if (not isinstance(fingerprint, str) or len(fingerprint) != 64
                    or any(c not in '0123456789abcdef' for c in fingerprint)
                    or rid in seen or len(seen) >= 4096):
                raise RuntimeError('Invalid or duplicate accepted frame identity')
            seen.add(rid)
            folder = root / fingerprint
            if folder.is_symlink() or not folder.resolve().is_relative_to(root.resolve()):
                raise RuntimeError('Unsafe accepted frame directory')
            available, missing = [], []
            for name in expected:
                path = folder / name
                relative = 'retained/' + fingerprint + '/' + name
                if path.is_symlink() or (path.exists() and not path.is_file()):
                    raise RuntimeError('Unsafe accepted frame file: ' + relative)
                (available if path.is_file() else missing).append(relative)
            producer = producer_by_id.get(rid, {})
            frames.append({'retention_id': rid, 'request_sha256': fingerprint,
                'window_number': window.get('number'), 'consumer_state': entry.get('state'),
                'consumer_reason': entry.get('reason'), 'producer_status': producer.get('status'),
                'producer_reason': producer.get('reason'), 'available_files': available, 'missing_files': missing})
    return {'accepted_count': len(frames), 'frames': frames,
        'missing_accepted_frames': [row for row in frames if row['missing_files']],
        'not_stored_accepted_frames': [row for row in frames if row['producer_status'] != 'stored'],
        'scope': 'Presence inventory only, not hashes, receipt validity, or a passing strict audit. Missing expected files are never fabricated.'}


def native_delta(a, b):
    return {'xy_distance_m': math.hypot(b['x'] - a['x'], b['y'] - a['y']),
            'yaw_delta_rad': delta(b['yaw'], a['yaw'])}


def writer_transport(checkpoint, shutdown):
    """Summarize only actually recorded unique writer operations, never infer success."""
    indexed, conflicts = {}, []
    keys = ('pid', 'operation_id', 'job_id', 'started_wall', 'returncode', 'stop_reason',
            'forced_kill', 'termination_unconfirmed', 'fault', 'stdin_pipe', 'input_transfer')
    for source, snapshot in (('checkpoint', checkpoint), ('shutdown', shutdown)):
        for index, raw in enumerate(snapshot.get('writer_history', [])):
            identity = raw.get('operation_id') or source + ':' + str(index)
            row = {key: raw.get(key) for key in keys}
            if identity in indexed:
                if indexed[identity]['recorded'] != row:
                    conflicts.append({'operation_id': identity, 'source': source, 'recorded': row})
                indexed[identity]['sources'].append(source)
            else:
                indexed[identity] = {'sources': [source], 'recorded': row}
    rows = list(indexed.values())
    for entry in rows:
        row = entry['recorded']
        transfer = row.get('input_transfer') or {}
        started, first, closed = row.get('started_wall'), transfer.get('first_progress_wall'), transfer.get('stdin_closed_wall')
        numeric = lambda value: type(value) in (int, float) and math.isfinite(value)
        entry['owner_sample_start_to_first_progress_s'] = first - started if numeric(first) and numeric(started) else None
        entry['owner_sample_start_to_stdin_closed_s'] = closed - started if numeric(closed) and numeric(started) else None
    def known_remaining(entry):
        value = (entry['recorded'].get('input_transfer') or {}).get('remaining_bytes')
        return type(value) in (int, float) and math.isfinite(value) and value >= 0
    stopped = [entry for entry in rows if entry['recorded'].get('stop_reason')]
    return {'recorded_unique_operation_count': len(rows),
        'operations_with_pipe_diagnostics': sum(isinstance(e['recorded'].get('stdin_pipe'), dict) for e in rows),
        'actual_pipe_capacity_counts': dict(Counter(str((e['recorded'].get('stdin_pipe') or {}).get('actual_bytes')) for e in rows)),
        'tuning_error_counts': dict(Counter(str((e['recorded'].get('stdin_pipe') or {}).get('tuning_error')) for e in rows)),
        'stop_reason_counts': dict(Counter(str(e['recorded'].get('stop_reason')) for e in rows)),
        'writer_wall_timeout_operations': [e for e in rows if e['recorded'].get('stop_reason') == 'writer_wall_timeout'],
        'operations_with_unsent_bytes_at_recorded_stop': [e for e in stopped if known_remaining(e)
            and e['recorded']['input_transfer']['remaining_bytes'] > 0],
        'unknown_remaining_bytes_at_recorded_stop': [e for e in stopped if not known_remaining(e)],
        'operations': rows, 'conflicting_recorded_versions': conflicts,
        'scope': 'Union of available bounded writer_history snapshots, deduplicated by operation_id; successful checkpoint-only workers are not normally included and earlier jobs may have rolled out. Times are owner samples, not child read/import/fsync timings. Unknown is not zero; no whole-run throughput or root-cause claim.'}


def main():
    if OUT.with_suffix('.json').exists() or OUT.with_suffix('.md').exists():
        raise RuntimeError('Refuse overwriting an outcome review')
    orchestration = load('orchestration.json', required=True)
    token_cleanup = load('run-token-final-processes.json', required=True)
    shutdown = load('retention-shutdown.json', required=True)
    if not (orchestration.get('status') in ('terminal_observed', 'orchestration_failed')
            and type(orchestration.get('elapsed_wall_s')) in (int, float)
            and math.isfinite(orchestration['elapsed_wall_s']) and orchestration['elapsed_wall_s'] > 0
            and orchestration.get('all_owned_groups_empty') is True
            and shutdown.get('writer') is None
            and shutdown.get('shutdown', {}).get('owned_writer_exit_confirmed') is True
            and shutdown.get('shutdown', {}).get('final_checkpoint_persisted') is True):
        raise RuntimeError('Run closure, owned groups, and producer writer exit not confirmed')
    # A global /proc permission failure does not preclude reading closed files.
    # Keep its original failure and ancillary rechecks separate: never promote
    # an incomplete global token scan to proven process absence.
    token_recheck = load('run-token-process-recheck.json')
    denied_context = load('denied-process-context.json')
    terminal = orchestration.get('terminal') or {}
    # Resolve the recorded mission filename within this run, also on Windows.
    name = str(terminal.get('path', '')).replace('\\', '/').rsplit('/', 1)[-1]
    mission_source = 'supervisor_terminal'
    mission_shutdown = {}
    if not name:
        shutdowns = sorted((RUN / 'mission').glob('shutdown_*.json'))
        if len(shutdowns) != 1:
            raise RuntimeError('No exact unique mission shutdown evidence')
        name = shutdowns[0].name.replace('shutdown_', 'mission_', 1)
        mission_shutdown = load(Path('mission') / shutdowns[0].name, required=True)
        if not (mission_shutdown.get('goal_terminal_verified') is True
                and mission_shutdown.get('stop_verified') is True):
            raise RuntimeError('Mission cancellation/stop closure not verified')
        mission_source = 'unique_shutdown_filename_pair; not_a_supervisor_terminal'
    if not name.startswith('mission_') or not name.endswith('.json'):
        raise RuntimeError('No exact terminal mission path in the supervisor report')
    mission = load(Path('mission') / name, required=True)
    if mission.get('status') not in ('done', 'blocked'):
        raise RuntimeError('Saved mission is not terminal')
    observer = load('observer.json')
    audit = load('retention-audit.json')
    checkpoint = load('retained/checkpoint.json', required=True)
    if not (checkpoint.get('closed') is True and checkpoint.get('writer') is None
            and checkpoint.get('checkpoint_failed') is False
            and checkpoint.get('core', {}).get('closed') is True
            and checkpoint.get('core', {}).get('writer_pending') is False
            and checkpoint.get('fault') == shutdown.get('fault')):
        raise RuntimeError('Final checkpoint closure or preserved fault disagrees with shutdown')
    for snapshot in (checkpoint, shutdown):
        if any(type(row.get('returncode')) is not int or row.get('termination_unconfirmed') is not False
               for row in snapshot.get('writer_history', [])):
            raise RuntimeError('A recorded writer lacks exact exit confirmation')
    preparation = load('preparation.json')
    final = load('final-save.json')
    raster = load('snapshot-map.provenance.json')
    raster_failure = load('snapshot-map.failure.json')
    events = mission.get('events', [])
    tf_rejections = [e for e in events if e.get('event') == 'base_tf_age_rejected']
    extrapolation = navigation_extrapolation()
    goals = [e for e in events if e.get('event') == 'goal_sent']
    terminals = [e for e in events if e.get('event') == 'goal_terminal']
    terminal_by_token = {e['token']: e for e in terminals}
    dispatches = [{**e, 'terminal': terminal_by_token.get(e['token'])} for e in goals]
    kinds = {}
    for kind in sorted({e['goal']['kind'] for e in goals}):
        rows = [e for e in dispatches if e['goal']['kind'] == kind]
        kinds[kind] = {'sent': len(rows),
                      'terminal_success': sum((r['terminal'] or {}).get('success') is True for r in rows),
                      'terminal_failure': sum((r['terminal'] or {}).get('success') is False for r in rows),
                      'without_recorded_terminal': sum(r['terminal'] is None for r in rows)}
    actions = mission.get('observation_actions') or {}
    attempts = actions.get('attempts', [])
    consumer = mission.get('frame_retention') or {}
    producer_results = checkpoint.get('core', {}).get('results', [])
    producer_by_id = {r['retention_id']: r for r in producer_results}
    audited_by_id = {r['retention_id']: r for r in audit.get('frames', [])}
    availability = accepted_file_inventory(consumer, producer_by_id)
    if (availability['missing_accepted_frames'] or availability['not_stored_accepted_frames']) and audit.get('passed') is True:
        raise RuntimeError('Passing strict audit contradicts incomplete accepted-frame evidence')
    transport = writer_transport(checkpoint, shutdown)
    windows = []
    for window in consumer.get('windows', []):
        frames = []
        for entry in window.get('accepted', []):
            request = entry['request']
            rid = request['retention_id']
            source = producer_by_id.get(rid, {})
            seen = entry.get('result') or {}
            frames.append({'retention_id': rid, 'accepted_ordinal': request['accepted_ordinal'],
                'request_sha256': entry['request_sha256'], 'consumer_state': entry.get('state'),
                'consumer_reason': entry.get('reason'), 'producer_status': source.get('status'),
                'producer_reason': source.get('reason'),
                'producer_request_digest_matches': bool(source) and source.get('request_sha256') == entry['request_sha256'],
                'consumer_checkpoint_persisted': seen.get('checkpoint_persisted'),
                'offline_audit': audited_by_id.get(rid)})
        windows.append({key: window.get(key) for key in (
            'number', 'binding', 'state', 'reason', 'processing_satisfied', 'ledger_accepted_count', 'late_results')}
            | {'accepted_count': len(frames), 'frames': frames})
    world = observer.get('world_samples', [])
    home_event = next((e for e in events if e.get('event') == 'home_recorded'), None)
    native_return = {'scope': 'Same native model sampled at home and final observer time; no map/world identity assumption.'}
    if world and home_event:
        before = [p for p in world if p['sim_s'] <= home_event['stamp']]
        after = [p for p in world if p['sim_s'] >= home_event['stamp']]
        if before and after:
            left, right = max(before, key=lambda p: p['sim_s']), min(after, key=lambda p: p['sim_s'])
            movement = native_delta(left, right)
            same_id = left['id'] == right['id'] == world[-1]['id']
            stationary = same_id and movement['xy_distance_m'] <= 1e-9 and abs(movement['yaw_delta_rad']) <= 1e-9
            native_return.update(home_event_stamp=home_event['stamp'], home_bracket=[left, right],
                home_bracket_gap_s=right['sim_s'] - left['sim_s'], home_bracket_movement=movement,
                same_native_id=same_id, home_bracket_stationary_at_sample_endpoints=stationary,
                final_sample=world[-1],
                sampled_home_to_final=native_delta(left, world[-1]) if stationary else None)
    home, pose = mission.get('home'), mission.get('final_pose')
    map_return = None if home is None or pose is None else {
        'xy_distance_m': math.hypot(pose[0] - home[0], pose[1] - home[1]),
        'yaw_error_rad': abs(delta(pose[2], home[2]))}
    flags = ('status', 'reason', 'mission_success', 'returned_pose_verified', 'map_complete',
             'visual_coverage_complete', 'mission_timed_out', 'target_definition_verified', 'coordinates_valid')
    report = {'schema_version': 1, 'role': 'offline_run27_outcome_summary',
        'run': RUN.name, 'runtime_head': preparation.get('code_head'), 'map_epoch': mission.get('map_epoch'),
        'previous_attempt': preparation.get('previous_attempt'),
        'experiment_profile': {key: preparation.get(key) for key in ('change', 'writer_profile', 'comparison_scope')},
        'validation_provenance': {'code_head': preparation.get('code_head'),
            'scope': 'Revision and profile read from original run27 preparation; available installed-check and test/build receipts are separately archived. No fixed runtime SHA or validation success is assumed.'},
        'mission': {key: mission.get(key) for key in flags},
        'mission_evidence_source': mission_source, 'mission_shutdown': mission_shutdown,
        'supervisor_terminal_recorded': bool(terminal),
        'tf_diagnostics': {'base_tf_age_rejected_event_count': len(tf_rejections),
            'base_tf_age_rejected_events': tf_rejections, 'navigation_extrapolation': extrapolation,
            'scope': 'Node logs only its first attempted age-rejection diagnostic; counts do not measure all rejected lookups and logger failure may omit an event. Decision now/age and event-emission stamp are distinct.'},
        'navigation': {'by_kind': kinds, 'dispatches': dispatches,
            'cancel_events': [e for e in events if e.get('event') == 'cancel_requested'],
            'note': 'Terminal failure includes cancellation; timeout classification uses explicit recorded reasons only.'},
        'inspection': {'completed': [e for e in events if e.get('event') == 'inspection_complete'],
            'failed': [e for e in events if e.get('event') == 'inspection_failed'],
            'interrupted': [e for e in events if e.get('event') == 'inspection_interrupted']},
        'observation': {'snapshot': actions, 'finished_outcomes': dict(Counter(a.get('outcome') for a in attempts)),
            'explicit_timeout_attempts': [a for a in attempts if 'timeout' in str(a.get('reason', ''))]},
        'retention': {'enabled': consumer.get('enabled'), 'consumer_fault': consumer.get('fault'),
            'accepted_count': sum(w['accepted_count'] for w in windows), 'windows': windows,
            'accepted_file_inventory': availability,
            'producer_result_counts': dict(Counter(r.get('status') for r in producer_results)),
            'producer_checkpoint_closed': checkpoint.get('closed'), 'producer_fault': checkpoint.get('fault'),
            'shutdown': shutdown, 'offline_audit': audit,
            'original_offline_audit_available': bool(audit),
            'audit_scope': 'Existing closed-run audit result, not a second raw-image decode or independent TF-accuracy test.'},
        'writer_transport': transport,
        'targets': {'count': len(mission.get('targets', [])), 'annotations': mission.get('targets', []),
            'unconfirmed': mission.get('unconfirmed_target_candidates'), 'count_prior': mission.get('target_count_prior')},
        'return': {'home_map_pose': home, 'final_map_pose': pose, 'map_error': map_return,
            'native': native_return, 'stationary_tail': observer.get('final_native_tail_30sim'),
            'runner_full_terminal_tail_verified': orchestration.get('native_terminal_tail_verified')},
        'observer': {'errors': observer.get('errors'), 'final_status': observer.get('final_status'),
            'native_sample_count': len(world), 'native_ids': sorted({p['id'] for p in world})},
        'map_export': {'snapshot_and_posegraph': {key: final.get(key) for key in
            ('passed', 'errors', 'exports', 'exported_files_at_capture', 'raster_export_mode', 'save_map_service_attempted')},
            'separate_snapshot_raster': raster, 'separate_snapshot_raster_failure': raster_failure,
            'terminal_evidence_collection': orchestration.get('terminal_evidence_collection'),
            'scope': 'Separate captured OccupancyGrid raster and posegraph service; not a save_map service repair.'},
        'cleanup': orchestration, 'run_token_cleanup': token_cleanup,
        'run_token_cleanup_context': {'original_global_absence_verified':
            orchestration.get('all_run_token_processes_absent'),
            'recheck': token_recheck, 'denied_process_context': denied_context,
            'scope': 'Ancillary evidence only; does not override the original incomplete global scan.'},
        'limitations': ['No ROS nodes or motion commands started by this review; no truth files read.',
            'Owned groups and the retained-frame writer have closure evidence; global token absence remains as reported, including permission failures.',
            'A bounded queue of processed-frame windows does not establish whole-maze visual coverage or target recall.',
            'Saved online claims and closed-run artifact audit have different scopes; missing evidence is not a pass.',
            'Native stationary brackets/tails are sampled evidence, not a continuous-motion proof.',
            'No physical LIMO hardware evaluated.'], 'missing_files': MISSING, 'source_files': SOURCES}
    with OUT.with_suffix('.json').open('x', encoding='utf-8') as stream:
        json.dump(report, stream, ensure_ascii=False, indent=2, allow_nan=False)
        stream.write('\n')
    lines = ['# run27 离线结果摘要', '', '终态与原始证据闭合后生成；不读取目标真值，不启动 ROS。', '',
        f"运行版本：`{preparation.get('code_head')}`；配置与测试/构建回执分开保留，不预设结果。", '',
        '```json', json.dumps(report['mission'], ensure_ascii=False, indent=2), '```', '',
        '| 导航类型 | 下发 | 成功终态 | 失败终态（含取消） | 无记录终态 |', '|---|---:|---:|---:|---:|']
    for kind, counts in kinds.items():
        lines.append('| ' + kind + ' | ' + ' | '.join(str(v) for v in counts.values()) + ' |')
    lines += ['', f"观察动作结果：`{json.dumps(report['observation']['finished_outcomes'], ensure_ascii=False)}`。",
        f"TF age 首次拒绝事件：{len(tf_rejections)}；navigation extrapolation 匹配日志行数：{extrapolation.get('matched_line_count')}。",
        'TF 事件保留判据 now、TF stamp、age、阈值与 frames；日志行计数不是独立故障数，不用零条日志推断 TF 无故障。',
        f"关闭原因：`{actions.get('close_reason')}`；显式 timeout 原因的观察动作数：{len(report['observation']['explicit_timeout_attempts'])}。", '',
        f"保存的 consumer 原始接受数：{report['retention']['accepted_count']}；离线文件审计："
        f"passed={audit.get('passed')}，verified={audit.get('verified_count')}/{audit.get('accepted_count')}。", '',
        f"保留的 producer fault：`{checkpoint.get('fault')}`；缺少预期文件的 accepted 帧数：{len(availability['missing_accepted_frames'])}；producer 未 stored 的 accepted 帧数：{len(availability['not_stored_accepted_frames'])}。",
        '缺失 ID 与逐文件路径见 JSON accepted_file_inventory；文件存在性不等于严格审计通过。严格审计可能因 producer fault 提前返回 failed 0/0，其计数不替代 mission 原始接受数。', '',
        f"有记录的唯一 writer 操作：{transport['recorded_unique_operation_count']}；具有管道诊断：{transport['operations_with_pipe_diagnostics']}；"
        f"实际容量分布：`{json.dumps(transport['actual_pipe_capacity_counts'])}`；"
        f"writer 超时记录：{len(transport['writer_wall_timeout_operations'])}。",
        '逐操作初始/请求/实际容量、扩容错误、发送/剩余字节、EAGAIN、首末进展及关闭时间、最大 poll 间隔均在 writer_transport。历史有长度上限，不代表全部任务；父端时间不等于子进程读取或写盘时间。', '',
        f"地图内标注 {report['targets']['count']} 个目标；返航地图误差：`{json.dumps(map_return)}`。", '',
        f"原生 home 括点到最终采样：`{json.dumps(native_return.get('sampled_home_to_final'))}`；"
        '只有相同模型且 home 前后采样位姿相同时计算，详见 JSON 括点时间与跨度。', '',
        f"记录的尾窗：`{json.dumps(report['return']['stationary_tail'])}`。",
        f"编排器完整终态尾窗验证：`{orchestration.get('native_terminal_tail_verified')}`；"
        '采样跨度及是否全部处于终态后以原始字段为准，不把部分尾窗当成完整通过。', '',
        f"编排器状态：`{orchestration.get('status')}`；错误：`{json.dumps(orchestration.get('errors'), ensure_ascii=False)}`。",
        f"地图快照与 posegraph 捕获：passed=`{final.get('passed')}`；错误：`{json.dumps(final.get('errors'), ensure_ascii=False)}`。",
        f"save_map 服务是否调用：`{final.get('save_map_service_attempted')}`；独立捕获栅格导出：passed=`{raster.get('passed')}`。",
        '独立栅格导出与原生终态尾窗结果分别保留，不互相替代，也不称为修复 save_map 服务。', '',
        '所有具体阈值、失败原因、accepted ID、producer 状态、原始审计及来源 SHA256 见同名 JSON。', '',
        '**本摘要不把 done、有限多朝向观察或原图留存通过解释为全迷宫视觉覆盖、正式目标识别或实车验收。**', '']
    with OUT.with_suffix('.md').open('x', encoding='utf-8') as stream:
        stream.write('\n'.join(lines))
    print(json.dumps({'json': str(OUT.with_suffix('.json')), 'markdown': str(OUT.with_suffix('.md')),
                      'mission': report['mission'], 'retention_accepted': report['retention']['accepted_count'],
