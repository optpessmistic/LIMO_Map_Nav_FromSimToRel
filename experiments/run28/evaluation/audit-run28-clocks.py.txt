"""Offline stamps from the closed run28 bag; no ROS node or playback."""
from collections import defaultdict
import hashlib
import json
from pathlib import Path
import re
import rosbag2_py
from rclpy.serialization import deserialize_message
from rosidl_runtime_py.utilities import get_message

ROOT = Path(__file__).resolve().parents[2]
RUN = ROOT / 'outputs/ros_smoke/run28-startup-timing'
OUT = ROOT / 'outputs/setup/run28-clock-audit.json'
if OUT.exists():
    raise RuntimeError('Refuse overwriting audit')
closed = json.loads((RUN / 'orchestration.json').read_bytes())
assert closed['all_owned_groups_empty'] and closed['elapsed_wall_s'] > 0
bag = RUN / 'recording/bag/bag_0.mcap'
sha = hashlib.sha256(bag.read_bytes()).hexdigest()
assert sha == 'b099a1585a7868f6c978bfc117a4ef145ee3fe31837fcf0a503fd08de1769b33'
reader = rosbag2_py.SequentialReader()
reader.open(rosbag2_py.StorageOptions(uri=str(bag.parent), storage_id='mcap'),
            rosbag2_py.ConverterOptions('', ''))
topics = {x.name: get_message(x.type) for x in reader.get_all_topics_and_types()
          if x.name in ('/clock', '/tf', '/odom', '/scan')}
reader.set_filter(rosbag2_py.StorageFilter(topics=list(topics)))
series = defaultdict(list)
def ns(stamp):
    return stamp.sec * 10**9 + stamp.nanosec
while reader.has_next():
    name, raw, stamp = reader.read_next()
    msg = deserialize_message(raw, topics[name])
    if name == '/tf':
        for tf in msg.transforms:
            series[tf.header.frame_id + '->' + tf.child_frame_id].append((stamp, ns(tf.header.stamp)))
    else:
        series[name].append((stamp, ns(msg.clock if name == '/clock' else msg.header.stamp)))
navigation = (RUN / 'navigation.log').read_text()
future = []
for line_no, line in enumerate(navigation.splitlines(), 1):
    match = re.search(r'Requested time ([0-9.]+) but the latest data is at time ([0-9.]+)', line)
    if match:
        requested, latest = map(float, match.groups())
        future.append({'line': line_no, 'requested_sim_s': requested,
                       'latest_sim_s': latest, 'difference_s': requested - latest, 'text': line})
result = {'run': RUN.name, 'bag_sha256': sha, 'read_only': True, 'topics': {},
          'navigation_future_extrapolation_lines': future, 'event_windows': [],
          'limitations': [
              'Bag timestamps use the recorder ROS clock, not independent wall receipt time.',
              'This cannot reconstruct the mission or Nav2 private TF buffers, scheduling, or prove a unique clock publisher.',
              'SLAM can future-date map->odom by transform_timeout; that alone is not a mixed-clock fault.',
              'Repeated log lines are not necessarily independent incidents; windows selected from requested ROS times.']}
distinct_series = {}
for name, rows in series.items():
    ages = [(b-s)/1e9 for b, s in rows if s > 0]
    distinct = []
    for row in rows:
        if not distinct or row[1] != distinct[-1][1]:
            distinct.append(row)
    gaps = [(y[0]-x[0])/1e9 for x, y in zip(distinct, distinct[1:])]
    distinct_series[name] = distinct
    result['topics'][name] = {
        'count': len(rows), 'header_first_s': rows[0][1]/1e9, 'header_last_s': rows[-1][1]/1e9,
        'min_header_s': min(r[1] for r in rows)/1e9, 'max_header_s': max(r[1] for r in rows)/1e9,
        'header_backwards_count': sum(y[1] < x[1] for x, y in zip(rows, rows[1:])),
        'bag_minus_header_min_s': min(ages) if ages else None,
        'bag_minus_header_max_s': max(ages) if ages else None,
        'max_bag_time_between_header_changes_s': max(gaps) if gaps else None}
for event in future:
    center = event['requested_sim_s']
    result['event_windows'].append({'requested_sim_s': center, 'half_width_sim_s': 1,
        'stamp_changes': {name: [{'bag_ros_s': b/1e9, 'header_s': s/1e9, 'bag_minus_header_s': (b-s)/1e9}
            for b, s in rows if center-1 <= b/1e9 <= center+1]
            for name, rows in distinct_series.items() if name != '/clock'}})
capture = json.loads((RUN / 'parameters-early.json').read_bytes())
result['parameter_receipts'] = {node: {'status': row['status'], 'use_sim_time': row['values'].get('use_sim_time')}
    for node, row in capture['parameters'].items()}
assert hashlib.sha256(bag.read_bytes()).hexdigest() == sha
with OUT.open('x') as stream:
    json.dump(result, stream, indent=2, allow_nan=False)
    stream.write('\n')
print(json.dumps({'output': str(OUT), 'topics': result['topics'],
                  'future_extrapolation_lines': len(future)}, indent=2))
