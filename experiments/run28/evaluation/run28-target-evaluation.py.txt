#!/usr/bin/env python3
"""DRAFT: independent offline target scoring for a CLOSED normal mission.

Required CLI: --run-dir --alignment --world --metadata --output.
Missing normal terminal mission or closure refuses output. Missing/invalid
alignment or truth produces target_evaluation=null and an explicit reason;
no synthetic zero-target result is made. No ROS/Gazebo imports or commands.

Uses the post-survey/pre-first-goal native-root alignment schema generated by
freeze-run28-initial-alignment.py. Frozen TF/native records must match the
hashed observer source exactly; the first eligible stamp defines the gauge.
It does not accept the older run14 sensor-reference schema or fit predictions.
The separately validated model/SDF provenance remains the alignment producer's
responsibility; this consumer checks its declared model reference, source
hashes, live static TF, raw observer association and planar reconstruction.

Prediction reference is an RGB-D visible surface point; truth is the red-box
centre. Matching is one-to-one, same-class, planar XY within 0.20 m. There is no
surface-to-centre correction, Z scoring, or aggregate benchmark verdict.
"""
import argparse
import bisect
from datetime import datetime, timezone
import importlib.util
import json
import math
from pathlib import Path
import sys
import xml.etree.ElementTree as ET

import numpy as np

ROOT = Path(__file__).resolve().parents[2]
GEOMETRY_PATH = Path(__file__).with_name('run28-geometric-coverage.py')
spec = importlib.util.spec_from_file_location('run28_geometry_helpers', GEOMETRY_PATH)
geometry = importlib.util.module_from_spec(spec)
spec.loader.exec_module(geometry)  # Definitions only; its guarded main is not called.
sys.path.insert(0, str(ROOT / 'tools'))
from evaluate_targets import SE2, evaluate  # noqa: E402

require, finite, integer = geometry.require, geometry.finite, geometry.integer
PHASE = 'post_initial_survey_before_first_autonomous_goal'
CLASS_MAP = {'red_box_surrogate': 'red_object'}


def wrap(value):
    return math.atan2(math.sin(value), math.cos(value))


def pose_matrix(xyz, quaternion):
    xyz = geometry.vector(xyz, 3, 'TF.xyz')
    x, y, z, w = geometry.vector(quaternion, 4, 'TF.quaternion_xyzw')
    require(abs(x*x+y*y+z*z+w*w-1) <= 1e-6, 'TF quaternion is not unit')
    result = np.eye(4)
    result[:3, :3] = [[1-2*(y*y+z*z), 2*(x*y-z*w), 2*(x*z+y*w)],
                      [2*(x*y+z*w), 1-2*(x*x+z*z), 2*(y*z-x*w)],
                      [2*(x*z-y*w), 2*(y*z+x*w), 1-2*(x*x+y*y)]]
    result[:3, 3] = xyz
    return result


def rigid_matrix(value):
    require(type(value) is list and len(value) == 4, 'Model reference matrix must have four rows')
    result = np.array([geometry.vector(row, 4, 'model_reference.matrix') for row in value])
    rotation = result[:3, :3]
    require(np.max(np.abs(result[3]-[0, 0, 0, 1])) <= 1e-9
            and np.max(np.abs(rotation.T @ rotation-np.eye(3))) <= 1e-6
            and abs(float(np.linalg.det(rotation))-1) <= 1e-6, 'Invalid model-reference rigid matrix')
    return result


def verified_manifest_source(path, scope):
    actual = geometry.fingerprint(path)
    matches = [item for item in scope['source_inputs_verified']
               if item.get('path') == actual['path'] and 'prefix_bytes' not in item]
    require(len(matches) == 1 and matches[0] == actual, 'Exact run source absent from frozen alignment: ' + str(path))
    return geometry.load(path)


def raw_alignment_checks(alignment, scope, run, mission):
    require(alignment.get('offline_only') is True and alignment.get('calibration_phase') == PHASE,
            'Expected offline post-survey/pre-first-goal alignment')
    epoch = mission['map_epoch']
    model = alignment['model_reference']
    require(model.get('native_entity') == 'limo' and model.get('urdf_root') == 'base_footprint'
            and model.get('sdf_canonical_root_identity_verified') is True
            and model.get('runtime_static_tf_matches_submitted_urdf') is True,
            'Native root/model reference was not independently declared validated')
    native_id = integer(model.get('native_id'), 'native_id')
    root_base = rigid_matrix(model['base_footprint_to_base_link_matrix'])
    runtime = verified_manifest_source(run / 'sim-runtime.json', scope)
    require(runtime.get('passed') is True, 'Runtime static TF precheck failed')
    tf = runtime['tf_chain']['base_footprint->base_link']
    require(np.max(np.abs(pose_matrix(tf['xyz'], tf['quaternion_xyzw'])-root_base)) <= 1e-9,
            'Model-reference transform differs from hashed runtime static TF')
    survey = verified_manifest_source(run / 'initial-survey.json', scope)
    require(survey.get('success') is True and survey.get('map_epoch') == epoch, 'Original survey success/epoch mismatch')
    observer = verified_manifest_source(run / 'observer.json', scope)
    require(observer.get('world_entity') == 'limo', 'Original observer native entity mismatch')
    events = mission.get('events')
    require(isinstance(events, list), 'Mission event list required')
    goals = [finite(event['stamp'], 'goal_sent.stamp') for event in events if event.get('event') == 'goal_sent']
    require(bool(goals), 'No autonomous goal in normal terminal mission')
    first_goal = min(goals)
    window = alignment['validation_window']
    require(finite(window.get('first_goal_sim_s'), 'first_goal_sim_s') == first_goal
            and window.get('all_calibration_samples_before_first_goal') is True,
            'Frozen calibration timing differs from actual first autonomous goal')
    native = observer['world_samples']
    require(isinstance(native, list) and len(native) >= 2, 'Missing raw native world samples')
    native_stamps = [round(finite(item['sim_s'], 'native.sim_s') * 1e9) for item in native]
    require(all(a < b for a, b in zip(native_stamps, native_stamps[1:])), 'Nonmonotonic raw native stamps')
    raw_rows = observer['continuous_tf_samples']
    require(isinstance(raw_rows, list), 'Raw observer TF rows required')
    eligible = []
    for index, row in enumerate(raw_rows):
        ns = row.get('reference_stamp_ns')
        if (row.get('map_epoch') != epoch or row.get('same_stamp_chain_complete') is not True
                or type(ns) is not int or ns <= 0 or ns*1e-9 >= first_goal
                or finite(row.get('ros_s', first_goal), 'TF.ros_s') >= first_goal
                or row.get('active_goal_token') is not None):
            continue
        if native_stamps[0] <= ns <= native_stamps[-1]:
            eligible.append((ns, index))
    eligible.sort()
    require(bool(eligible), 'No actual eligible pre-goal TF/native anchor')
    first_ns = eligible[0][0]
    expected, seen = [], set()
    for ns, index in eligible:
        if ns <= first_ns+2_000_000_000 and ns not in seen:
            expected.append((ns, index))
            seen.add(ns)
    frames = alignment['validation_frames']
    require(isinstance(frames, list) and len(frames) >= 3
            and [(frame['stamp_ns'], frame['observer_tf_index']) for frame in frames] == expected
            and expected[-1][0]-first_ns >= 200_000_000,
            'Frozen frames are not the first eligible anchor and complete bounded validation window')
    require(alignment['reference'] == frames[0] and alignment['reference_stamp_ns'] == first_ns
            and window.get('first_stamp_ns') == first_ns and window.get('last_stamp_ns') == expected[-1][0]
            and window.get('frame_count') == len(frames), 'Reference/window fields disagree with raw frame selection')
    frozen = SE2(**alignment['map_to_world'])
    reconstructed, native_poses = [], []
    inverse_root_base = np.linalg.inv(root_base)
    for frame in frames:
        ns = integer(frame['stamp_ns'], 'frame.stamp_ns', 1)
        index = integer(frame['observer_tf_index'], 'observer_tf_index', 0, len(raw_rows)-1)
        row = raw_rows[index]
        require(frame['source_tf_row'] == row and row.get('map_epoch') == epoch
                and row.get('reference_stamp_ns') == ns and row.get('same_stamp_chain_complete') is True,
                'Frozen TF row differs from original same-epoch observer record')
        hi = bisect.bisect_left(native_stamps, ns)
        lo = hi if native_stamps[hi] == ns else hi-1
        bracket = frame['native_model_bracket']
        require(type(bracket.get('before_index')) is int and type(bracket.get('after_index')) is int
                and bracket['before_index'] == lo and bracket['after_index'] == hi
                and bracket['before'] == native[lo] and bracket['after'] == native[hi],
                'Frozen native bracket/index differs from original world samples')
        left, right = native[lo], native[hi]
        gap = (native_stamps[hi]-native_stamps[lo])*1e-9
        require(0 <= gap <= .1 and abs(finite(bracket['gap_s'], 'native.gap_s')-gap) <= 1e-12
                and left['id'] == right['id'] == native_id, 'Native identity/bracket duration mismatch')
        for sample in (left, right):
            for key in ('x', 'y', 'yaw'):
                finite(sample[key], 'native.' + key)
        require(math.hypot(left['x']-right['x'], left['y']-right['y']) <= 1e-9
                and abs(wrap(left['yaw']-right['yaw'])) <= 1e-9, 'Native bracket is not endpoint-stationary')
        matrices = {}
        for key, parent, child in (('map_to_base_link', 'map', 'base_link'),
                                  ('map_to_odom', 'map', 'odom'), ('odom_to_base_link', 'odom', 'base_link')):
            tf = row['transforms'][key]
            require(tf.get('stamp_ns') == ns and tf.get('parent') == parent and tf.get('child') == child,
                    'TF chain does not share the exact reference stamp/frames')
            matrices[key] = pose_matrix(tf['xyz'], tf['quaternion_xyzw'])
        require(np.max(np.abs(matrices['map_to_odom'] @ matrices['odom_to_base_link']-matrices['map_to_base_link'])) <= 1e-6,
                'Raw same-stamp TF chain fails composition')
        map_root = matrices['map_to_base_link'] @ inverse_root_base
        require(max(abs(map_root[2, 0]), abs(map_root[2, 1]), abs(map_root[0, 2]),
                    abs(map_root[1, 2]), abs(map_root[2, 2]-1)) <= 1e-6, 'Map-to-root orientation is not planar')
        map_yaw = math.atan2(map_root[1, 0], map_root[0, 0])
        require(np.max(np.abs(map_root[:3, 3]-geometry.vector(frame['map_to_base_footprint_xyz'], 3, 'map root XYZ'))) <= 1e-9
                and abs(wrap(map_yaw-finite(frame['map_to_base_footprint_yaw'], 'map root yaw'))) <= 1e-9,
                'Stored map root differs from raw TF and explicit root/base transform')
        yaw = wrap(left['yaw']-map_yaw)
        rx, ry = SE2(0, 0, yaw).transform(*map_root[:2, 3])
        measured = SE2(left['x']-rx, left['y']-ry, yaw)
        saved = SE2(**frame['map_to_world'])
        require(math.hypot(measured.x-saved.x, measured.y-saved.y) <= 1e-9
                and abs(wrap(measured.yaw-saved.yaw)) <= 1e-9, 'Stored per-frame SE2 differs from raw reconstruction')
        reconstructed.append(measured)
        native_poses.append(left)
    require(math.hypot(frozen.x-reconstructed[0].x, frozen.y-reconstructed[0].y) <= 1e-9
            and abs(wrap(frozen.yaw-reconstructed[0].yaw)) <= 1e-9, 'Frozen SE2 is not the first raw anchor')
    max_xy = max(math.hypot(item.x-frozen.x, item.y-frozen.y) for item in reconstructed)
    max_yaw = max(abs(wrap(item.yaw-frozen.yaw)) for item in reconstructed)
    base = native_poses[0]
    native_xy = max(math.hypot(item['x']-base['x'], item['y']-base['y']) for item in native_poses)
    native_yaw = max(abs(wrap(item['yaw']-base['yaw'])) for item in native_poses)
    require(max_xy <= .001 and max_yaw <= .001 and native_xy <= .001 and native_yaw <= .001,
            'Reconstructed pre-goal validation exceeds 1 mm/.001 rad')
    return {'phase': PHASE, 'reference_stamp_ns': first_ns, 'first_goal_sim_s': first_goal,
        'validation_frame_count': len(frames), 'last_validation_stamp_ns': expected[-1][0],
        'raw_observer_identity': geometry.fingerprint(run / 'observer.json'), 'model_reference': model,
        'max_map_to_world_xy_variation_m': max_xy, 'max_map_to_world_yaw_variation_rad': max_yaw,
        'native_world_xy_span_m': native_xy, 'native_world_yaw_span_rad': native_yaw,
        'scope': 'Exact raw observer/native association and explicit root/base planar reconstruction; submitted URDF/SDF derivation remains separately validated by the frozen alignment producer.'}


def unchanged(sources, prefixes):
    for item in sources.values():
        require(geometry.fingerprint(item['path']) == item, 'Input changed during offline scoring: ' + item['path'])
    for item in prefixes:
        require(geometry.fingerprint(item['path'], item['prefix_bytes'])['sha256'] == item['prefix_sha256'],
                'Alignment prefix changed during offline scoring')


def main():
    parser = argparse.ArgumentParser(description=__doc__, formatter_class=argparse.RawDescriptionHelpFormatter)
    for name in ('run-dir', 'alignment', 'world', 'metadata', 'output'):
        parser.add_argument('--' + name, type=Path, required=True)
    args = parser.parse_args()
    require(not args.output.exists(), 'Refuse overwriting target evidence')
    geometry.SOURCES.clear()
    run = args.run_dir.resolve(strict=True)
    require(run.is_dir(), 'Run directory required')
    mission, orchestration = geometry.closed_mission(run)  # Hard refusal, outside the nullable scoring branch.
    closed_sources = dict(geometry.SOURCES)
    for path in (Path(__file__), GEOMETRY_PATH, ROOT / 'tools/evaluate_targets.py'):
        geometry.remember(path)
    target_evaluation, alignment_scope, raw_checks, truth_scope, unverified_reason = None, None, None, None, None
    prefixes = []
    try:
        preparation = geometry.load(run / 'preparation.json')
        world = geometry.remember(args.world)
        require(world['sha256'] == preparation.get('world_sha256'), 'Supplied world differs from run preparation')
        transform, alignment_scope, prefixes = geometry.checked_alignment(args.alignment, mission['map_epoch'], world['sha256'])
        alignment = geometry.load(args.alignment)
        raw_checks = raw_alignment_checks(alignment, alignment_scope, run, mission)
        require(mission.get('frame_id') == 'map', 'Target mission coordinates must use the validated map frame')
        metadata = geometry.load(args.metadata)
        boxes, bounds, spawn = geometry.truth_geometry(args.world, metadata)
        truth_scope = {'world': world, 'metadata': geometry.fingerprint(args.metadata),
            'validated_target_box_count': sum(item['name'].startswith('target_surrogate_') for item in boxes),
            'domain_bounds_xy': bounds, 'metadata_spawn_world_xy': spawn,
            'scope': 'Prepared static world hash and matching collision/visual box centres/sizes; no independent attestation of all live spawned geometry.'}
        alignment_identity = geometry.fingerprint(args.alignment)
        target_evaluation = evaluate(mission, metadata, alignment=SE2(**transform),
            alignment_source=str(args.alignment) + '; sha256=' + alignment_identity['sha256'],
            appearance_to_class=CLASS_MAP, max_distance_m=.20)
        unchanged(geometry.SOURCES, prefixes)
    except (ValueError, KeyError, TypeError, IndexError, AttributeError, OSError, ET.ParseError, OverflowError, np.linalg.LinAlgError) as exc:
        target_evaluation = None
        unverified_reason = type(exc).__name__ + ': ' + str(exc)
    unchanged(closed_sources, [])  # Never publish a result from a changed terminal/closure record.
    original = {key: mission.get(key) for key in ('schema_version', 'status', 'reason', 'mission_success',
        'failures', 'map_complete', 'visual_coverage_complete', 'returned_pose_verified', 'coordinates_valid',
        'target_definition_verified', 'mission_timed_out')}
    report = {'schema_version': 1, 'role': 'draft_offline_run28_target_evaluation', 'run': run.name,
        'map_epoch': mission['map_epoch'], 'original_mission': original,
        'target_evaluation': target_evaluation, 'target_evaluation_unverified_reason': unverified_reason,
        'alignment': alignment_scope, 'raw_alignment_checks': raw_checks, 'truth': truth_scope,
        'created_at_utc': datetime.now(timezone.utc).isoformat(),
        'metric_definition': {'prediction_reference': 'RGB-D visible surface point in map transformed by fixed post-survey gauge',
            'truth_reference': 'Wall-mounted red surrogate box centre', 'distance_metric': 'planar_xy_euclidean_m',
            'max_distance_m': .20, 'appearance_to_class': CLASS_MAP, 'surface_to_centre_correction_applied': False,
            'z_evaluated': False, 'matching': 'Maximum-cardinality, then minimum-total-distance, one-to-one same-class matches'},
        'original_orchestration_status': orchestration.get('status'), 'original_orchestration_errors': orchestration.get('errors'),
        'global_process_absence_verified': orchestration.get('all_run_token_processes_absent'),
        'inputs': geometry.SOURCES, 'alignment_prefix_inputs': prefixes,
        'execution': {'python': sys.executable, 'argv': sys.argv, 'cwd': str(Path.cwd())},
        'limits': ['Original mission_success, failures and other mission flags are unchanged; no overall benchmark verdict is inferred.',
            'Null target_evaluation means missing/invalid evidence, not zero targets or a failed detector.',
            'Post-survey alignment cannot measure SLAM drift during initial Spin and cannot be compared as the same gauge as run14 pre-Spin alignment.',
            'No target position, terminal pose or final map is used to fit an alignment; world truth is offline only.',
            'Target scoring does not validate full exploration, visual coverage, autonomy, collision-free operation, return home, formal items or physical LIMO.']}
    encoded = json.dumps(report, ensure_ascii=False, indent=2, allow_nan=False) + '\n'
    with args.output.open('x', encoding='utf-8') as stream:
        stream.write(encoded)
    print(json.dumps({'output': str(args.output), 'run': run.name,
                      'target_scored': target_evaluation is not None, 'unverified_reason': unverified_reason}, ensure_ascii=False))


if __name__ == '__main__':
    try:
        main()
    except (ValueError, TypeError, KeyError, OSError, OverflowError) as exc:
        raise SystemExit('REFUSED: ' + str(exc))
