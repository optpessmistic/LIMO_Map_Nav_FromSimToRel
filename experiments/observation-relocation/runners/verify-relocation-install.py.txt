import hashlib
import importlib
import json
from pathlib import Path

root = Path(__file__).resolve().parents[2]
out = root / 'outputs/validation/observation-relocation-build'
receipt = {'colcon_exit_code': 0,
           'initial_wrapper_exit_code': 1,
           'initial_wrapper_error': "ModuleNotFoundError: No module named 'limo_maze_mission'",
           'initial_wrapper_scope': 'colcon completed successfully; importing in the pre-build parent environment failed. Fresh shell sourced the post-build install for this independent verification.',
           'installed_modules': {}}
for name in ('core', 'observation', 'node'):
    module = importlib.import_module('limo_maze_mission.' + name)
    installed = Path(module.__file__)
    expected = root / 'src/limo_maze_mission/limo_maze_mission' / f'{name}.py'
    actual_sha = hashlib.sha256(installed.read_bytes()).hexdigest()
    source_sha = hashlib.sha256(expected.read_bytes()).hexdigest()
    receipt['installed_modules'][name] = {
        'path': str(installed), 'resolved': str(installed.resolve()),
        'sha256': actual_sha, 'source_sha256': source_sha,
        'matches_source': actual_sha == source_sha,
    }
code = 0 if all(entry['matches_source'] for entry in receipt['installed_modules'].values()) else 1
receipt['fresh_environment_verification_exit_code'] = code
with (out / 'result.json').open('x', encoding='utf-8') as stream:
    stream.write(json.dumps(receipt, indent=2) + '\n')
print(json.dumps(receipt, indent=2), flush=True)
raise SystemExit(code)
