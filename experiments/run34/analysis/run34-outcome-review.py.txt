"""Summarize closed run34 evidence without reinterpreting its success flags.

Do not execute until root session 22438 has actually completed. This tool starts
no ROS/Gazebo processes and does not audit pixels again. Existing outputs are
never overwritten; a failed original audit remains failed.
"""
import argparse
from collections import Counter, defaultdict
import hashlib
import json
import math
from pathlib import Path

ROOT = Path(__file__).resolve().parents[2]
RUN = ROOT/'outputs/ros_smoke/run34-lifecycle-readiness'
DEFAULT_OUT = ROOT/'outputs/setup/run34-outcome-review'
EXPECTED_EPOCH = 'ea53a5de-b919-4f52-aa28-ea6d8ec571c8'
EXPECTED_TOKEN = 'run34-fbc7f96364474fc49e89f384d770cc76'
EXPECTED_HEAD = 'c411fcbe86c1c27ce302593b73fba405a603e002'
INPUTS, MISSING, STATS = {}, [], {}


def obj(value):
    return value if isinstance(value, dict) else {}


def finite(value):
    return type(value) in (int, float) and math.isfinite(value)


def pick(value, keys):
    return {key:obj(value).get(key) for key in keys.split()}


def read_bytes(path, required=False):
    path = Path(path)
    if not path.exists():
        if required:
            raise RuntimeError('Required evidence missing: '+str(path))
        MISSING.append(str(path.relative_to(ROOT)))
        return None
    if path.is_symlink() or not path.is_file() or not path.resolve().is_relative_to(ROOT.resolve()):
        raise RuntimeError('Unsafe input: '+str(path))
    before = path.stat()
    signature = lambda st: (st.st_dev, st.st_ino, st.st_size, st.st_mtime_ns)
    if path in STATS and signature(before) != STATS[path]:
        raise RuntimeError('Previously read evidence changed: '+str(path))
    if before.st_size > 512*1024*1024:
        raise RuntimeError('Input exceeds bounded reader: '+str(path))
    raw = path.read_bytes()
    if signature(before) != signature(path.stat()) or len(raw) != before.st_size:
        raise RuntimeError('Input changed while reading; do not review an active run')
    name = path.relative_to(ROOT).as_posix()
    INPUTS[name] = {'bytes':len(raw),'sha256':hashlib.sha256(raw).hexdigest()}
    STATS[path] = signature(before)
    return raw


def load(name, required=False):
    raw = read_bytes(RUN/name, required)
    return None if raw is None else json.loads(raw)


def validate_closed(orchestration, preparation, args):
    if not (orchestration.get('run') == RUN.name and preparation.get('run') == RUN.name
            and preparation.get('epoch') == EXPECTED_EPOCH and preparation.get('token') == EXPECTED_TOKEN
            and preparation.get('code_head') == preparation.get('prepared_head_actual') == EXPECTED_HEAD
            and orchestration.get('epoch') == preparation.get('epoch')
            and orchestration.get('status') in ('terminal_observed','orchestration_failed')
            and orchestration.get('stage') == 'owned_shutdown'
            and finite(orchestration.get('elapsed_wall_s')) and orchestration['elapsed_wall_s'] > 0
            and type(orchestration.get('all_owned_groups_empty')) is bool
            and 'evidence_collection_complete' in orchestration):
        raise RuntimeError('Final same-run supervisor receipt missing; root must first confirm completion')
    if read_bytes(RUN/'epoch.txt', True) != (preparation['epoch']+'\n').encode('ascii'):
        raise RuntimeError('Preparation epoch bytes disagree')
    stops = obj(orchestration.get('stops'))
    nested = obj(orchestration.get('recorder_inner_recheck'))
    if 'recorder' in stops and not (nested.get('remaining_check_complete') is True
            and nested.get('remaining') == [] and nested.get('ownership_errors') == []):
        raise RuntimeError('Nested recorder closure unconfirmed; outer group check is insufficient')
    if orchestration['all_owned_groups_empty'] is True:
        stops = orchestration.get('stops')
        if not (isinstance(stops,dict) and stops and all(
                r.get('remaining_check_complete') is True and r.get('remaining') == []
                and r.get('ownership_errors') == [] for r in stops.values())):
            raise RuntimeError('Owned closure flag contradicts original stop records')
        return None
    if args.later_closure_check is None:
        raise RuntimeError('Original owned closure failed/uncertain; provide a separate later closure receipt before reading remaining evidence')
    later_path = Path(args.later_closure_check).resolve()
    later = json.loads(read_bytes(later_path, True))
    completion, groups = obj(later.get('root_exec_completion')), later.get('groups')
    original_sha = INPUTS[(RUN/'orchestration.json').relative_to(ROOT).as_posix()]['sha256']
    if not (later.get('run') == RUN.name and later.get('epoch') == preparation['epoch']
            and later.get('original_orchestration_sha256') == original_sha
            and later.get('original_all_owned_groups_empty') is False
            and later.get('all_owned_groups_empty_at_later_check') is True
            and later.get('signals_sent') == [] and isinstance(groups,list) and groups
            and completion.get('session_id') == args.confirmed_root_session
            and completion.get('exit_code') == args.confirmed_root_exit_code):
        raise RuntimeError('Later closure receipt does not bind this original failed run')
    names = []
    for row in groups:
        owner_path = ROOT/row['owner_file']
        raw = read_bytes(owner_path, True)
        if not (owner_path.parent == RUN and hashlib.sha256(raw).hexdigest() == row.get('owner_sha256')
                and json.loads(raw) == row.get('owner') and row['owner'].get('token') == preparation['token']
                and row.get('remaining') == [] and row.get('errors') == []
                and row.get('remaining_check_complete') is True):
            raise RuntimeError('Later owned-group identity or closure disagrees')
        names.append(row['owner']['name'])
    if len(set(names)) != len(names) or set(names) != set(obj(orchestration.get('stops'))):
        raise RuntimeError('Later closure receipt omits/duplicates an original group')
    return {'path':later_path.relative_to(ROOT).as_posix(),
        'sha256':hashlib.sha256(read_bytes(later_path,True)).hexdigest(),
        'owned_group_count':len(groups),'original_all_owned_groups_empty':False,
        'all_owned_groups_empty_at_later_check':True,
        'scope':'Historical later check only; original failure and global process uncertainty unchanged.'}


def mission_evidence(orchestration, preparation):
    terminal = orchestration.get('terminal')
    if not terminal:
        return None, 'No hash-bound supervisor terminal; mission statistics remain unknown.'
    name = str(terminal.get('path','')).replace('\\','/').rsplit('/',1)[-1]
    paths = sorted((RUN/'mission').glob('mission_*.json'))
    if not name.startswith('mission_') or not name.endswith('.json') or [p.name for p in paths] != [name]:
        raise RuntimeError('Supervisor terminal does not identify a unique saved mission')
    raw = read_bytes(RUN/'mission'/name, True)
    mission = json.loads(raw)
    if not (hashlib.sha256(raw).hexdigest() == terminal.get('sha256')
            and mission.get('map_epoch') == terminal.get('map_epoch') == preparation['epoch']
            and mission.get('status') == terminal.get('status') and mission['status'] in ('done','blocked')
            and mission.get('mission_success') == terminal.get('reported_mission_success')):
        raise RuntimeError('Terminal mission hash/epoch/status mismatch')
    return mission, 'Exact unique supervisor terminal filename, raw SHA256 and epoch verified.'


def navigation_summary(events):
    if events is None:
        return {'by_kind':None,'dispatches':None,'terminal_errors':None}
    sends = [e for e in events if e.get('event') == 'goal_sent']
    ends = [e for e in events if e.get('event') == 'goal_terminal']
    end_by_token = defaultdict(list)
    for end in ends:
        end_by_token[end.get('token')].append(end)
    rows, by_kind = [], {}
    send_tokens = Counter(e.get('token') for e in sends)
    for sent in sends:
        token, goal = sent.get('token'), obj(sent.get('goal'))
        terminals = end_by_token.get(token, [])
        terminal = terminals[0] if len(terminals) == 1 and send_tokens[token] == 1 else None
        kind = goal.get('kind') or 'unknown_kind'
        counts = by_kind.setdefault(kind, {'sent':0,'terminal_success':0,'terminal_failure':0,'terminal_unknown':0})
        counts['sent'] += 1
        success = obj(terminal).get('success')
        counts['terminal_success' if success is True else 'terminal_failure' if success is False else 'terminal_unknown'] += 1
        rows.append({'token':token,'goal':goal,'stamp':sent.get('stamp'),
                     'terminal':terminal,'recorded_terminal_count':len(terminals)})
    return {'by_kind':by_kind,'dispatches':rows,
        'terminal_errors':[pick(e,'token stamp success status error_code error_msg detail') for e in ends],
        'recovery_started_count':sum(e.get('event') == 'navigation_recovery_started' for e in events),
        'recovery_ready_count':sum(e.get('event') == 'navigation_recovery_ready' for e in events),
        'scope':'Recorded sends joined by unique token; failure includes cancellation. Duplicate/absent terminals stay unknown.'}


def observation_summary(mission, navigation):
    actions = obj(obj(mission).get('observation_actions'))
    queue, counts = obj(actions.get('queue')), obj(obj(actions.get('queue')).get('state_counts'))
    planned = queue.get('planned_views')
    known_counts = ('state_counts' in queue and isinstance(queue['state_counts'],dict)
                    and all(type(v) is int and v >= 0 for v in counts.values()))
    total = sum(counts.values()) if known_counts else None
    attempts = actions.get('attempts')
    rows = list(attempts) if isinstance(attempts,list) else []
    if isinstance(actions.get('active'),dict) and actions['active']:
        rows.append(actions['active'])
    nav = navigation.get('dispatches')
    details = []
    for attempt in rows:
        goal, planned_goal = obj(attempt.get('goal')), obj(attempt.get('planned_goal'))
        matches = [r for r in nav or [] if r['token'] == attempt.get('nav_token') and r['goal'] == goal]
        sent = None if nav is None or len(matches) > 1 else (len(matches) == 1
                and finite(attempt.get('dispatched_wall'))
                and type(attempt.get('nav_token')) is int and attempt['nav_token'] > 0)
        coordinates = [planned_goal.get(k) for k in ('x','y')] + [goal.get(k) for k in ('x','y')]
        shift = math.hypot(coordinates[2]-coordinates[0],coordinates[3]-coordinates[1]) if all(map(finite,coordinates)) else None
        selection = obj(attempt.get('selection'))
        details.append({**pick(attempt,'action_id dispatch_id station_id heading_index nav_token outcome reason nav_terminal_status'),
            'goal_send_verified':sent,'planned_goal':planned_goal or None,'actual_sent_goal':goal if sent else None,
            'planned_to_selected_xy_m':shift,'shifted_goal_sent':sent and shift > 1e-9 if shift is not None else None,
            'selection':pick(selection,'map relocation_radius_m relocation_reason selected_shift_from_anchor_m original_check selected_check'),
            'inspection_satisfied_as_recorded':attempt.get('outcome') == 'inspection_satisfied'})
    shifted = [r for r in details if r['shifted_goal_sent'] is True]
    known_relocation_count = isinstance(attempts,list) and nav is not None and all(
        r['goal_send_verified'] is not None and (not r['goal_send_verified']
            or r['planned_to_selected_xy_m'] is not None) for r in details)
    return {'queue_field_recorded':'queue' in actions,
        'queue_created':isinstance(actions.get('queue'),dict) if 'queue' in actions else None,
        'planned_views_denominator':planned if type(planned) is int and planned >= 0 else None,
        'state_counts_as_recorded':queue.get('state_counts'), 'state_count_total':total,
        'counts_partition_planned_views':total == planned if known_counts and type(planned) is int else None,
        'completed':counts.get('completed',0) if known_counts else None,
        'skipped_unsafe_or_unreachable':counts.get('skipped_unsafe_or_unreachable',0) if known_counts else None,
        'failed':counts.get('failed',0) if known_counts else None,
        'pending':counts.get('pending',0) if known_counts else None,
        'active':counts.get('active',0) if known_counts else None,
        'all_planned_actions_completed':queue.get('all_planned_actions_completed'),
        'planned_stations':queue.get('planned_stations'),'dispatches_as_recorded':queue.get('dispatches'),
        'closed':actions.get('closed'),'close_reason':actions.get('close_reason'),
        'selection_history':queue.get('selection_history'),
        'attempts':details if isinstance(attempts,list) else None,
        'shifted_goal_sent_count':len(shifted) if known_relocation_count else None,
        'verified_shifted_goal_sent_lower_bound':len(shifted) if nav is not None and isinstance(attempts,list) else None,
        'unknown_goal_send_verification_count':sum(r['goal_send_verified'] is None for r in details) if isinstance(attempts,list) else None,
        'shifted_goal_inspection_satisfied_count':sum(r['inspection_satisfied_as_recorded'] for r in shifted) if known_relocation_count else None,
        'unknown_planned_to_selected_shift_count':sum(r['planned_to_selected_xy_m'] is None for r in details) if isinstance(attempts,list) else None,
        'scope':'Actual queue denominator includes skipped/failed/pending/active. Relocation counts shifted goal sends, not physical displacement or arrival. Inspection completion is separate; neither implies visual coverage.'}


def retention_summary(mission, audit, checkpoint, shutdown):
    consumer, core = obj(obj(mission).get('frame_retention')), obj(obj(checkpoint).get('core'))
    windows = consumer.get('windows')
    known = isinstance(windows,list) and all(isinstance(w,dict) and isinstance(w.get('accepted'),list) for w in windows)
    accepted = [a for w in windows for a in w['accepted']] if known else []
    audit_rows = obj(audit).get('frames')
    verified = [r for r in audit_rows if r.get('verified') is True] if isinstance(audit_rows,list) else None
    matched = sum(any(r.get('retention_id') == obj(a.get('request')).get('retention_id')
                      and r.get('request_sha256') == a.get('request_sha256') for r in verified)
                  for a in accepted) if known and verified is not None else None
    results = core.get('results')
    return {'consumer_accepted_count':len(accepted) if known else None,
        'consumer_stored_state_count':sum(a.get('state') == 'stored' for a in accepted) if known else None,
        'consumer_fault':consumer.get('fault'),'producer_fault':obj(checkpoint).get('fault'),
        'producer_result_status_counts':dict(Counter(r.get('status') for r in results)) if isinstance(results,list) else None,
        'checkpoint_closed':obj(checkpoint).get('closed'),
        'original_audit':pick(audit,'passed accepted_count verified_count errors scope'),
        'verified_audit_frame_rows':len(verified) if verified is not None else None,
        'accepted_entries_matching_verified_audit_identity':matched,
        'shutdown':shutdown,
        'scope':'Original strict audit of reopened accepted RGB-D files; verified_count counts frame pairs, not independent file objects. No repeat pixel audit. Failed 0/0 audit never replaces a nonzero consumer ledger or becomes a pass.'}


def native_return(observer, events):
    result = {'sampled_home_to_final':None,'reason':None,
        'scope':'Native Gazebo world samples only; no map/world identity assumption. Distance is measured displacement from the home bracket, not a return acceptance result. Equal bracket endpoints are sampled evidence, not continuous stationarity.'}
    samples = obj(observer).get('world_samples')
    homes = [e for e in events or [] if e.get('event') == 'home_recorded']
    if not isinstance(samples,list) or not samples or len(homes) != 1 or not finite(homes[0].get('stamp')):
        result['reason'] = 'Missing native samples or unique recorded home timestamp.'
        return result
    if any(not isinstance(p,dict) or p.get('id') is None or not all(finite(p.get(k)) for k in ('sim_s','x','y','yaw')) for p in samples):
        result['reason'] = 'Malformed native sample; no filtered replacement used.'
        return result
    if any(b['sim_s'] < a['sim_s'] for a,b in zip(samples,samples[1:])):
        result['reason'] = 'Recorded native simulation time regressed.'
        return result
    stamp = homes[0]['stamp']
    before, after = [p for p in samples if p['sim_s'] <= stamp], [p for p in samples if p['sim_s'] >= stamp]
    if not before or not after:
        result['reason'] = 'Home timestamp lacks two-sided native sampling bracket.'
        return result
    left, right, final = before[-1], after[0], samples[-1]
    def delta(a,b):
        angle = b['yaw']-a['yaw']
        return {'xy_distance_m':math.hypot(b['x']-a['x'],b['y']-a['y']),
                'yaw_delta_rad':math.atan2(math.sin(angle),math.cos(angle))}
    movement = delta(left,right)
    same = left['id'] == right['id'] == final['id']
    stationary = same and movement['xy_distance_m'] <= 1e-9 and abs(movement['yaw_delta_rad']) <= 1e-9
    result.update(home_event_stamp=stamp,home_bracket=[left,right],final_sample=final,
        home_bracket_gap_s=right['sim_s']-left['sim_s'],home_bracket_movement=movement,
        same_native_id=same,home_bracket_stationary_at_sample_endpoints=stationary,
        sampled_home_to_final=delta(left,final) if stationary else None,
        reason=None if stationary else 'Native identity or sampled home stationarity is not verified.')
    return result


def tf_summary(events):
    reports = {}
    for name in ('navigation.log','mission.log'):
        raw = read_bytes(RUN/name)
        if raw is None:
            reports[name] = None
            continue
        lines = raw.decode('utf-8',errors='replace').splitlines()
        rules = {'extrapolation':'extrapolat','stale_base_tf':'stale_base_tf','failed_transform':'failed to transform'}
        reports[name] = {rule:{'matched_line_count':sum(needle in s.lower() for s in lines),
            'first_five_lines':[s[:1000] for s in lines if needle in s.lower()][:5]}
            for rule,needle in rules.items()}
        camera_lines = [{'line_number':i+1,'line':line} for i,line in enumerate(lines)
            if 'rejected target observation' in line.lower() and 'camera_optical_frame' in line
            and any(text in line.lower() for text in ('does not exist',"doesn't exist",'not exist','invalid frame id'))]
        home_line = next((i+1 for i,line in enumerate(lines) if 'home_recorded' in line),None)
        reports[name]['target_observation_camera_frame_unavailable'] = {
            'matched_line_count':len(camera_lines),'matching_lines':camera_lines,
            'first_home_recorded_log_line':home_line,
            'before_first_home_recorded_line_count':sum(row['line_number'] < home_line for row in camera_lines) if home_line is not None else None,
            'scope':'Target observation rejection because camera frame is absent; separate from stale base TF and navigation extrapolation. Ordering is file-line order, not a wall/ROS clock conversion.'}
    rejected = [e for e in events if e.get('event') == 'base_tf_age_rejected'] if events is not None else None
    return {'base_tf_age_rejected_count':len(rejected) if rejected is not None else None,
        'base_tf_age_rejected_events':rejected,'logs':reports,
        'scope':'Substring log-line counts are diagnostics, not unique faults. The node records only its first TF age rejection; missing events do not certify zero rejected lookups.'}

        'observation_station_spacing_m observation_max_stations observation_headings_per_station observation_max_dispatches '
