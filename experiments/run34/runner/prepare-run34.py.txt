def sha(data):
    return hashlib.sha256(data).hexdigest()


def read(path):
    return json.loads(path.read_bytes())


def replace_once(source, old, new):
    if source.count(old) != 1:
        raise RuntimeError('Expected exactly one frozen template anchor: ' + old[:120])
    return source.replace(old, new, 1)


def template(previous, name):
    path = SETUP / name
    raw = path.read_bytes()
    expected = previous['prepared_helper_sha256'][path.relative_to(ROOT).as_posix()]
    if sha(raw) != expected:
        raise RuntimeError('Frozen run33 helper changed: ' + name)
    return raw.decode('utf-8').replace('\r\n', '\n')


def slam_snapshot():
    build = read(SLAM / 'build-source.json')
    source = SLAM / 'src/slam_toolbox'
    common = source / 'src/slam_toolbox_common.cpp'
    raw = common.read_bytes()
    if (sha(raw) != COMMON_SHA256
            or raw.count(b'rclcpp::WallRate r(1.0 / transform_publish_period);') != 1
            or raw.count(b'rclcpp::WallRate r(1.0 / map_update_interval);') != 1
            or b'rclcpp::Rate r(1.0 / transform_publish_period);' in raw
            or b'rclcpp::Rate r(1.0 / map_update_interval);' in raw):
        raise RuntimeError('SLAM does not have the exact pinned steady TF/map source')
    if sha((source/'lib/karto_sdk/include/karto_sdk/Karto.h').read_bytes()) != KARTO_SHA256:
        raise RuntimeError('Pinned inclusive-grid source changed')
    if build.get('patches') != PATCH_SHA256:
        raise RuntimeError('Build must bind exactly patches0002+0003+0004')
    files = {}
    for name, expected in build['source_files'].items():
        path = source / name
        if not path.resolve(strict=True).is_relative_to(source.resolve(strict=True)):
            raise RuntimeError('Source snapshot escapes new overlay: ' + name)
        actual = sha(path.read_bytes())
        if actual != expected:
            raise RuntimeError('Build source receipt mismatch: ' + name)
        files[path.relative_to(ROOT).as_posix()] = actual
    for name, expected in build['patches'].items():
        if sha((ROOT / 'patches' / name).read_bytes()) != expected:
            raise RuntimeError('Built patch differs from current input: ' + name)
    prefix = SLAM / 'install/slam_toolbox'
    libraries = {}
    for name in ('libtoolbox_common.so', 'libkartoSlamToolbox.so'):
        path = prefix / 'lib' / name
        resolved = path.resolve(strict=True)
        if not resolved.is_relative_to(SLAM.resolve(strict=True)):
            raise RuntimeError('New SLAM library resolves outside the new overlay: ' + name)
        checksum = sha(path.read_bytes())
        files[path.relative_to(ROOT).as_posix()] = checksum
        libraries[name] = {'path': str(path), 'resolved_path': str(resolved), 'sha256': checksum}
    for path in (SLAM / 'build-source.json', SLAM / 'install/local_setup.bash'):
        files[path.relative_to(ROOT).as_posix()] = sha(path.read_bytes())
    return {'overlay': str(SLAM), 'package_prefix': str(prefix),
            'source_common_sha256': sha(raw),
            'source_karto_header_sha256': sha((source / 'lib/karto_sdk/include/karto_sdk/Karto.h').read_bytes()),
            'libraries': libraries, 'files_sha256': files,
            'tf_stamp_policy': 'Unchanged ROS scan stamp plus transform_timeout; restamp_tf=false.',
            'scheduling_change': 'Relative to run32, only map publication scheduling changes Rate to WallRate; steady TF and scan-derived header semantics remain.'}




def receipt_path(value):
    path = Path(value)
    if not path.is_absolute():
        path = ROOT/path
    if path.is_symlink() or not path.is_file() or not path.resolve().is_relative_to(ROOT.resolve()):
        raise RuntimeError('Missing or unsafe receipt: '+str(path))
    return path.resolve()


def check_ci(path, expected_head):
    path = receipt_path(path)
    ci = read(path)
    if not (ci.get('head_sha') == expected_head and ci.get('status') == 'completed'
            and ci.get('conclusion') == 'success' and type(ci.get('id')) is int and ci['id'] > 0):
        raise RuntimeError('CI must be completed success for the exact expected source HEAD')
    return {'path':path.relative_to(ROOT).as_posix(),'sha256':sha(path.read_bytes()),**ci}


def build_validation(binding):
    """Bind root's closed receipt summaries without inferring raw internal schemas."""
    path = receipt_path(BUILD_RECEIPT)
    receipt = read(path)
    current_libraries = {name:row['sha256'] for name,row in binding['libraries'].items()}
    if not (receipt.get('exec_session_id') == 22504 and receipt.get('exit_code') == 0
            and receipt.get('status') == 'completed' and receipt.get('packages_finished') == 1
            and receipt.get('new_overlay') == SLAM.relative_to(ROOT).as_posix()
            and receipt.get('source_common_sha256') == COMMON_SHA256
            and receipt.get('new_library_sha256') == current_libraries
            and receipt.get('selected_package_prefix') == binding['package_prefix']
            and receipt.get('old_overlay_library_hashes_match_baseline') is True
            and receipt.get('wallrate_constructor_reference_present') is True
            and receipt.get('ldd_missing_dependencies') is False):
        raise RuntimeError('Completed original build receipt does not match new overlay artifacts')
    incremental = receipt.get('incremental_recheck') or {}
    recheck_path = receipt_path(incremental.get('path',''))
    if not (recheck_path == RECHECK_RECEIPT.resolve()
            and sha(recheck_path.read_bytes()) == incremental.get('sha256')
            and incremental.get('exec_session_id') == 99394
            and incremental.get('exit_code') == 1 and incremental.get('passed') is False
            and incremental.get('build_executed') is False):
        raise RuntimeError('Original failed pre-build incremental recheck receipt changed')
    installation = receipt.get('installation_check') or {}
    install_path = receipt_path(installation.get('path',''))
    if not (sha(install_path.read_bytes()) == installation.get('sha256')
            and type(installation.get('exec_session_id')) is int and installation['exec_session_id'] > 0
            and installation.get('exit_code') == 0 and installation.get('passed') is True):
        raise RuntimeError('Required completed relocated-installation check receipt missing')
    return {'build_receipt':{'path':path.relative_to(ROOT).as_posix(),'sha256':sha(path.read_bytes()),
            'exec_session_id':receipt['exec_session_id'],'exit_code':receipt['exit_code']},
        'incremental_recheck':{**incremental,'path':recheck_path.relative_to(ROOT).as_posix()},
        'installation_check':{**installation,'path':install_path.relative_to(ROOT).as_posix()},
        'first_build_clock_skew_warning':receipt.get('first_build_clock_skew_warning'),
        'scope':'Original full build and completed relocated-installation check bound to raw receipts. Incremental precondition failure, build_executed=false and original clock-skew warning remain. Runtime loaded-library identity is checked separately; no simulation result or timing-cause claim.'}



def rename_current(value, previous, token=None):
    if token is not None:
        value = value.replace(previous['token'], token)
    return (value.replace(OLD.name, OUT.name).replace('run33','run34')
            .replace('Run33','Run34').replace('RUN33','RUN34'))


def render_helpers(previous, token):
    # Run33 already has relocation CLI/readback and legal exact-source symlinks.
    # Copy those mechanisms, never inject them a second time.
    helpers = {name:rename_current(template(previous,'run33-'+name),previous,token) for name in NAMES}
    if helpers['env.sh'].count('-p observation_relocation_radius_m:=0.15') != 1:
        raise RuntimeError('Existing relocation parameter must occur exactly once')
    process = helpers['process.py']
    for fragment in ("relocation_expected={'observation_relocation_radius_m':0.15,",
        "'robot_radius_m':0.25,'goal_arrival_margin_m':0.05,'observation_station_spacing_m':1.0}",
        "run('save-early',60,", "run('parameter-gate',45,", "run('installed-check',45,",
        "run('provenance',90,", 'wait_motion_startup(60)', 'deadline = began+1800',
        "'retention_worker_timeout_s':5.0,'retention_request_timeout_s':10.0",
        "'retention_shutdown_timeout_s':8.0", 'def check_isolation():', "['gz','service','-l']"):
        if process.count(fragment) != 1:
            raise RuntimeError('Preserved controller bound/readback not unique: '+fragment)
    for fragment in ('-p tf_max_age_s:=1.5','-p mission_timeout_s:=900.0','-p return_timeout_s:=240.0',
        '-p goal_timeout_s:=120.0','-p observation_stage_timeout_s:=360.0','-p observation_max_stations:=3',
        '-p observation_headings_per_station:=4','-p observation_max_dispatches:=12',
        '-p require_retained_inspection:=true','-p retention_enabled:=true','-p retention_max_jobs:=96',
        '-p navigation_failure_cooldown_s:=1.0','-p navigation_recovery_timeout_s:=3.0',
        '--duration 70 --motion-wait 90'):
        if helpers['env.sh'].count(fragment) != 1:
            raise RuntimeError('Runtime setting changed: '+fragment)
    for name, fragments in {
        'installed-check.py':('exact_source','expected_package_prefix','raw_import_path','resolved_path',
                             'prepared_source_sha256','launch_rows','slam_passed'),
        'provenance.py':('def check_slam_overlay(', 'matches_mapped_inode_and_device','run34_ownership_verified')}.items():
        if any(fragment not in helpers[name] for fragment in fragments):
            raise RuntimeError('Required copied identity gate missing: '+name)
    for name,value in helpers.items():
        if '\r' in value or 'run33' in value or 'RUN33' in value or 'outputs/slam-steady-ws' in value:
            raise RuntimeError('Stale run/overlay or non-LF helper: '+name)
        if name.endswith('.py'):
            ast.parse(value,filename='run34-'+name)
    return helpers


def render_start(previous, expected_head):
    start = rename_current(template(previous,'start-run33.sh'),previous)
    start = replace_once(start,"['session_id'] == 74954","['session_id'] == 21930")
    start = replace_once(start," == 'd4ba9bec3b2e177d6d348c072e5afa11a18a9889'"," == '"+expected_head+"'")
    start = replace_once(start,
        '# Recheck run32 original exit1 with successful owned closure; retain mission/audit failure.',
        '# Recheck run33 original startup failure/exit1 and owned closure; terminal/audit remain absent.')
    start = replace_once(start,
        "assert prior_closed['retention_audit_passed'] == report['previous_attempt']['retention_audit_passed']",
        "assert terminal is None and report['previous_attempt']['terminal_mission'] is None\n"
        "assert prior_closed.get('retention_audit_passed') is None\n"
        "assert report['previous_attempt']['retention_audit_passed'] is None\n"
        "assert report['previous_attempt']['retention_audit_status'] == 'not_run_no_terminal_mission'\n"
        "assert report['survey_source_sha256'] == module.SURVEY_SHA256\n"
        "assert hashlib.sha256((root/'scripts/initial_survey.py').read_bytes()).hexdigest() == module.SURVEY_SHA256")
    embedded = start.split("<<'PY'\n",1)[1].split('\nPY\n',1)[0]
    ast.parse(embedded,filename='start-run34.sh:embedded-python')
    if ('later_postclosure' in start or 'outputs/slam-steady-ws' in start or '\r' in start
            or "['exit_code'] == 1" not in start or '21930' not in start or 'run33' in start):
        # Only the explanatory comment above intentionally refers to historical run33.
        cleaned = start.replace('# Recheck run33 original startup failure/exit1 and owned closure; terminal/audit remain absent.','')
        if ('later_postclosure' in cleaned or 'outputs/slam-steady-ws' in cleaned or '\r' in cleaned
                or "['exit_code'] == 1" not in cleaned or '21930' not in cleaned or 'run33' in cleaned):
            raise RuntimeError('Generated start identity/closure invariant failed')
    return start


def previous_run_evidence():
    for name,expected in PREVIOUS_EVIDENCE_SHA256.items():

            raise RuntimeError('Frozen run33 evidence changed: '+name)
    provenance = read(ARCHIVE/'archive-provenance.json')
    rows = {row['source']:row for row in provenance['copies']}
    if len(rows) != len(provenance['copies']):
        raise RuntimeError('Archived run33 source mapping is ambiguous')
    def original_and_archived(relative):
        original = OLD/relative
        row = rows.get(original.relative_to(ROOT).as_posix()) or {}
        raw = original.read_bytes()
        if not (row.get('destination') == 'raw/'+relative and row.get('encoding') == 'raw'
                and sha(raw) == row.get('source_sha256') == row.get('archive_sha256')
                and len(raw) == row.get('source_bytes') == row.get('archive_bytes')
                and (ARCHIVE/row['destination']).read_bytes() == raw):
            raise RuntimeError('Original/archived run33 raw binding differs: '+relative)
        return raw
    previous = json.loads(original_and_archived('preparation.json'))
    closed = json.loads(original_and_archived('orchestration.json'))
    startup = read(ARCHIVE/'startup-failure.json')
    completion = startup.get('root_exec_completion') or {}
    elapsed = closed.get('elapsed_wall_s')
    if not (previous.get('run') == OLD.name and previous.get('code_head') == 'd4ba9bec3b2e177d6d348c072e5afa11a18a9889'
            and closed.get('run') == OLD.name and closed.get('epoch') == previous.get('epoch') == startup.get('epoch')
            and closed.get('status') == 'orchestration_failed' and closed.get('stage') == 'owned_shutdown'
            and closed.get('all_owned_groups_empty') is True and closed.get('terminal') is None
            and type(elapsed) in (int,float) and math.isfinite(elapsed) and elapsed > 0
            and completion.get('session_id') == 21930 and completion.get('exit_code') == 1):

    for key,value in startup['original_orchestration'].items():
        if closed.get(key) != value:
            raise RuntimeError('Archived startup summary differs from raw orchestration: '+key)
    stops, nested = closed.get('stops') or {}, closed.get('recorder_inner_recheck') or {}
    if not (len(stops) == 12 and all(r.get('remaining') == [] and r.get('ownership_errors') == []

            and nested.get('remaining') == [] and nested.get('ownership_errors') == []
            and nested.get('remaining_check_complete') is True):
        raise RuntimeError('Run33 12 owned groups/nested recorder closure incomplete')
    for name in stops:
        owner = json.loads(original_and_archived('process-'+name+'.json'))
        if owner.get('name') != name or owner.get('token') != previous['token']:
            raise RuntimeError('Run33 owned process identity mismatch: '+name)

    if not (recorder.get('map_epoch') == previous['epoch'] and recorder.get('status') == 'stopped'

            and recorder.get('cleanup',{}).get('remaining') == []
            and recorder.get('cleanup',{}).get('remaining_check_complete') is True):
        raise RuntimeError('Run33 nested recorder receipt mismatch')
    writer = json.loads(original_and_archived('retention-shutdown.json'))
    checkpoint = json.loads(original_and_archived('retained/checkpoint.json'))
    if not (writer.get('writer') is None and writer.get('fault') is None

    if not (survey.get('map_epoch') == previous['epoch'] and survey.get('success') is False
            and survey.get('phase') == 'preflight' and survey.get('motion_started') is False
            and survey.get('spin_goals') == [] and survey.get('yaw_accumulated_rad') == 0
            and survey.get('max_displacement_m') == 0):

    if any(survey.get(key) != value for key,value in startup['survey_as_recorded'].items()):
        raise RuntimeError('Archived survey summary differs from raw survey')

def source_snapshot(expected_head):
    if subprocess.check_output(['git','status','--porcelain','--untracked-files=normal'],cwd=ROOT):
        raise RuntimeError('Use a clean reviewed/committed source tree')
    head = subprocess.check_output(['git','rev-parse','HEAD'],cwd=ROOT,text=True).strip()
    if head != expected_head:
        raise RuntimeError('Actual HEAD differs from --expected-head')
    names = subprocess.check_output(['git','ls-files','-z','--',*SOURCE_ROOTS],cwd=ROOT)
    paths = sorted(name.decode('utf-8') for name in names.split(b'\0') if name)
    if not paths:
        raise RuntimeError('Tracked runtime source inventory missing')
    return head,{name:sha((ROOT/name).read_bytes()) for name in paths}


def main(argv=None):
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument('--expected-head',required=True)
    parser.add_argument('--ci-receipt',type=Path,required=True)
    parser.add_argument('--confirmed-previous-root-session',type=int,choices=[21930],required=True)
    parser.add_argument('--confirmed-previous-root-exit-code',type=int,choices=[1],required=True)
    args = parser.parse_args(argv)
    if args.expected_head != EXPECTED_HEAD:
        raise RuntimeError('Provide the fixed reviewed lifecycle-readiness commit as --expected-head')
    if sys.platform != 'linux':
        raise RuntimeError('Prepare in WSL so artifact paths match the runtime')
    targets = [SETUP/('run34-'+name) for name in NAMES]+[SETUP/'start-run34.sh']
    if (OUT.exists() or OUT.is_symlink() or (ROOT/'experiments/run34').exists()
            or any(p.exists() or p.is_symlink() for p in targets)):
        raise RuntimeError('Refuse existing run34 runtime/helpers/archive; never overwrite/restart')
    closed,writer,previous,prior_mission = previous_run_evidence()
    nav = (ROOT/'src/limo_maze_bringup/config/nav2_rpp_tf_experimental.yaml').read_bytes()
    world = (ROOT/'worlds/smoke_maze.world').read_bytes()
    slam_params = 'src/limo_maze_bringup/config/slam_smoke_experimental.yaml'
    if nav != (OLD/'nav2-rpp-03m.yaml').read_bytes() or sha(nav) != previous['nav2_sha256']:
        raise RuntimeError('Nav2/TF parameters changed from run33')
    if sha(world) != previous['world_sha256'] or sha((ROOT/slam_params).read_bytes()) != previous['prepared_source_sha256'][slam_params]:
        raise RuntimeError('World or SLAM parameter profile changed from run33')
    if sha((ROOT/'scripts/capture_motion_evidence.py').read_bytes()) != previous['collector_source_sha256_at_preparation']:
        raise RuntimeError('Motion collector changed from run33')
    if sha((ROOT/'scripts/initial_survey.py').read_bytes()) != SURVEY_SHA256:
        raise RuntimeError('Reviewed bounded lifecycle retry source changed')
    head,sources = source_snapshot(args.expected_head)
    ci = check_ci(args.ci_receipt,head)
    binding = slam_snapshot()
    build = build_validation(binding)
    if binding != previous['slam_overlay'] or build != previous['slam_build_validation']:
        raise RuntimeError('Existing steady-map overlay/build/install receipts changed from run33')
    token,epoch = 'run34-'+uuid.uuid4().hex,str(uuid.uuid4())
    helpers,start = render_helpers(previous,token),render_start(previous,head)
