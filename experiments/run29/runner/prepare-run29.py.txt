def sha(data):
    return hashlib.sha256(data).hexdigest()


def read(path):
    return json.loads(path.read_bytes())


def replace_once(source, old, new):
    if source.count(old) != 1:
        raise RuntimeError('Expected exactly one frozen template anchor: ' + old[:120])
    return source.replace(old, new, 1)


def template(previous, name):
    path = SETUP / name
    raw = path.read_bytes()
    expected = previous['prepared_helper_sha256'][path.relative_to(ROOT).as_posix()]
    if sha(raw) != expected:
        raise RuntimeError('Frozen run28 helper changed: ' + name)
    return raw.decode('utf-8').replace('\r\n', '\n')

def render_start(previous):
    start = template(previous, 'start-run28.sh')
    start = start.replace('run28-startup-timing', OUT.name).replace('run28', 'run29')
    start = replace_once(start,
        "assert report['previous_attempt']['original_all_owned_groups_empty'] is False\n"
        "assert report['previous_attempt']['later_scoped_groups_empty'] is True\n",
        "assert report['previous_attempt']['original_all_owned_groups_empty'] is True\n"
        "assert report['previous_attempt']['root_exec_completion']['session_id'] == 74367\n"
        "assert report['previous_attempt']['root_exec_completion']['exit_code'] == 1\n")
    start = replace_once(start, "assert report['code_head'] is None\n", """assert report['code_head'] is None
assert not subprocess.check_output(['git','status','--porcelain','--untracked-files=normal'])
assert subprocess.check_output(['git','rev-parse','HEAD'],text=True).strip() == report['prepared_head_actual']
for name, expected in report['prepared_source_sha256'].items():
    assert hashlib.sha256((root/name).read_bytes()).hexdigest() == expected, name
assert hashlib.sha256((root/'worlds/smoke_maze.world').read_bytes()).hexdigest() == report['world_sha256']
assert report['navigation_recovery_parameters'] == {'navigation_failure_cooldown_s':1.0,'navigation_recovery_timeout_s':3.0}
assert (out/'epoch.txt').read_bytes() == (report['epoch']+'\\n').encode('ascii')
assert report['writer_profile']['worker_wall_seconds'] == 5.0
assert report['writer_profile']['request_wall_seconds'] == 10.0
assert report['writer_profile']['shutdown_wall_seconds'] == 8.0
""")
    embedded = start.split("<<'PY'\n", 1)[1].split('\nPY\n', 1)[0]
    ast.parse(embedded, filename='start-run29.sh:embedded-python')
    return start


def previous_run_evidence():
    closed = read(OLD / 'orchestration.json')
    elapsed = closed.get('elapsed_wall_s')
    if not (closed.get('status') == 'terminal_observed' and closed.get('terminal')
            and type(elapsed) in (int, float) and math.isfinite(elapsed) and elapsed > 0
            and closed.get('all_owned_groups_empty') is True):
        raise RuntimeError('Run28 has not preserved its reviewed terminal and owned closure receipt')
    stops = closed.get('stops') or {}
    nested = closed.get('recorder_inner_recheck') or {}
    if not (stops and all(row.get('remaining') == [] and row.get('ownership_errors') == []
                        and row.get('remaining_check_complete') is True for row in stops.values())
            and nested.get('remaining') == [] and nested.get('ownership_errors') == []
            and nested.get('remaining_check_complete') is True):
        raise RuntimeError('Run28 recorded group/nested-recorder closure is incomplete')
    writer = read(OLD / 'retention-shutdown.json')
    if not (writer.get('writer') is None and writer.get('fault') is None
            and writer.get('shutdown', {}).get('owned_writer_exit_confirmed') is True
            and writer.get('shutdown', {}).get('final_checkpoint_persisted') is True):
        raise RuntimeError('Run28 producer did not confirm writer exit and final checkpoint')
    return closed, writer, read(OLD / 'preparation.json')


def source_snapshot():
    if subprocess.check_output(['git','status','--porcelain','--untracked-files=normal'], cwd=ROOT):
        raise RuntimeError('Commit the reviewed changes and use a clean tree before preparation')
    head = subprocess.check_output(['git','rev-parse','HEAD'], cwd=ROOT, text=True).strip()
    names = subprocess.check_output(['git','ls-files','-z','--',*SOURCE_ROOTS], cwd=ROOT)
    paths = sorted(name.decode('utf-8') for name in names.split(b'\0') if name)
    if not paths:
        raise RuntimeError('No tracked runtime source files found')
    return head, {name: sha((ROOT / name).read_bytes()) for name in paths}


def main(argv=None):
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument('--confirmed-previous-root-session', type=int, choices=[74367], required=True)
    parser.add_argument('--confirmed-previous-root-exit-code', type=int, choices=[1], required=True)
    args = parser.parse_args(argv)
    targets = [SETUP / ('run29-' + name) for name in NAMES] + [SETUP / 'start-run29.sh']
    if OUT.exists() or OUT.is_symlink() or any(path.exists() or path.is_symlink() for path in targets):
        raise RuntimeError('Refuse any existing run29 output/helper; no restart or overwrite')
    closed, writer, previous = previous_run_evidence()
    nav = (ROOT / 'src/limo_maze_bringup/config/nav2_rpp_tf_experimental.yaml').read_bytes()
    world = (ROOT / 'worlds/smoke_maze.world').read_bytes()
    if nav != (OLD / 'nav2-rpp-03m.yaml').read_bytes() or sha(nav) != previous['nav2_sha256']:
        raise RuntimeError('Nav2/TF profile changed from run28')
    if sha(world) != previous['world_sha256']:
        raise RuntimeError('Comparison world changed from run28')
    for path in ('src/limo_maze_mission/limo_maze_mission/node.py',
                 'src/limo_maze_mission/launch/mission.launch.py'):
        source = (ROOT / path).read_bytes()
        ast.parse(source, filename=path)
        if not all(name.encode('ascii') in source for name in RECOVERY):
            raise RuntimeError('Recovery parameter implementation missing: '+path)
    head, sources = source_snapshot()
    collector = (ROOT / 'scripts/capture_motion_evidence.py').read_bytes()
    if sha(collector) != previous['collector_source_sha256_at_preparation']:
        raise RuntimeError('Motion collector changed from run28')
    token, epoch = 'run29-' + uuid.uuid4().hex, str(uuid.uuid4())
    helpers, start = render_helpers(previous, token), render_start(previous)
    payloads = {SETUP / ('run29-' + name): value.encode('utf-8') for name, value in helpers.items()}
    payloads[SETUP / 'start-run29.sh'] = start.encode('utf-8')
    report = {**previous, 'run': OUT.name, 'code_head': None, 'prepared_head_actual': head,
        'token': token, 'epoch': epoch, 'navigation_recovery_parameters': RECOVERY,
        'prepared_source_sha256': sources,
        'prepared_helper_sha256': {path.relative_to(ROOT).as_posix(): sha(data) for path,data in payloads.items()},
        'preparation_script_sha256': sha(Path(__file__).read_bytes()),
        'previous_attempt': {'run': OLD.name, 'status': closed['status'], 'errors': closed['errors'],
            'original_all_owned_groups_empty': closed['all_owned_groups_empty'],
            'original_orchestration_sha256': sha((OLD / 'orchestration.json').read_bytes()),
            'retention_shutdown_sha256': sha((OLD / 'retention-shutdown.json').read_bytes()),
            'retention_audit_passed': closed.get('retention_audit_passed'),
            'all_run_token_processes_absent': closed.get('all_run_token_processes_absent'),
            'global_scan_limitation': closed.get('global_scan_limitation'),
            'owned_writer_exit_confirmed': writer['shutdown']['owned_writer_exit_confirmed'],
            'final_checkpoint_persisted': writer['shutdown']['final_checkpoint_persisted'],
            'scope': 'Original owned-group closure and writer receipt; no global process-absence claim or success rewrite.',
            'root_exec_completion': {'session_id': args.confirmed_previous_root_session,
                'exit_code': args.confirmed_previous_root_exit_code,
                'source': 'Root confirmation supplied as explicit prepare arguments.'}},
        'change': 'Only mission navigation_failure_cooldown_s=1.0 and navigation_recovery_timeout_s=3.0; add read-only local-costmap parameter receipts.',
        'startup_change': 'None relative to run28: motion collector outer readiness remains 60 s under global 1800 s.',
        'local_costmap_parameter_capture': {'node': '/local_costmap/local_costmap',
            'names': ['transform_tolerance','use_sim_time'], 'parameters_modified': False,
            'transform_tolerance_expected_value': None},
        'comparison_scope': 'Same run28 world, TF/RPP values, /mnt/c output, strict raw-frame retention and startup60. Mission900/return240/goal120/observation360/inspection12/writer5/request10/shutdown8 unchanged; only mission recovery values change.'}
    # All validation precedes exclusive writes. Interrupted preparation remains
    # visible and cannot be reused. The old run directory is never a write target.
    if source_snapshot() != (head, sources):
        raise RuntimeError('Source tree changed while preparing run29')
    OUT.mkdir(parents=True)
    (OUT / 'mission').mkdir()
    for path, data in ((OUT / 'epoch.txt', (epoch+'\n').encode('ascii')),
                       (OUT / 'nav2-rpp-03m.yaml', nav), *payloads.items(),
                       (OUT / 'preparation.json', (json.dumps(report,indent=2,allow_nan=False)+'\n').encode('utf-8'))):
        with path.open('xb') as stream:
            stream.write(data)
    print(json.dumps({'prepared':str(OUT),'epoch':epoch,'token':token,'prepared_head_actual':head,
                      'launched':False,'navigation_recovery_parameters':RECOVERY},indent=2))


if __name__ == '__main__':
    main()
