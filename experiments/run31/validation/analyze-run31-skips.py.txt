def grid(snapshot):
    m = snapshot['map']
    return Grid(m['width'], m['height'], m['resolution'], *m['origin'], tuple(snapshot['map_data']))


def pose(snapshot):
    return (*snapshot['pose']['xyz'][:2], snapshot['pose']['yaw'])


def blockers(g, cell, radius, initial):
    reach, rows = math.ceil(radius/g.resolution+.5), []
    for dy in range(-reach, reach+1):
        for dx in range(-reach, reach+1):
            distance = math.hypot(max(abs(dx)-.5, 0), max(abs(dy)-.5, 0))*g.resolution
            target = cell[0]+dx, cell[1]+dy
            if distance <= radius and g.value(target) != 0:
                rows.append({'cell': target, 'occupancy_value': g.value(target), 'inside': g.inside(target),
                    'disk_center_to_cell_rectangle_distance_m': distance,
                    'world_cell_center': g.to_world(target), 'initial_value_at_same_grid_index': initial.value(target),
                    'same_index_caveat': 'Origin differs: equal indices are not exactly equal world rectangles.'})
    return sorted(rows, key=lambda row: (row['disk_center_to_cell_rectangle_distance_m'], row['cell']))


def exact_world_endpoint(g, xy, radius):
    """Diagnostic only: same reserve, precise world center, nonfree rectangles."""
    dx, dy = xy[0]-g.origin_x, xy[1]-g.origin_y
    c, s = math.cos(g.origin_yaw), math.sin(g.origin_yaw)
    x, y = c*dx+s*dy, -s*dx+c*dy
    outside_distance = min(x, y, g.width*g.resolution-x, g.height*g.resolution-y)
    collisions, nearest = [], None
    for cell in g.cells():
        if g.value(cell) == 0:
            continue
        x0, y0 = cell[0]*g.resolution, cell[1]*g.resolution
        x1, y1 = x0+g.resolution, y0+g.resolution
        distance = math.hypot(max(x0-x, 0., x-x1), max(y0-y, 0., y-y1))
        row = {'cell': cell, 'occupancy_value': g.value(cell),
               'distance_m': distance, 'radius_minus_distance_m': radius-distance}
        if nearest is None or distance < nearest['distance_m']:
            nearest = row
        if distance <= radius:
            collisions.append(row)
    return {'diagnostic_only': True, 'world_center': xy, 'unchanged_radius_m': radius,
        'circle_clear': not collisions and outside_distance > radius,
        'outside_map_distance_m': outside_distance, 'outside_map_intersects': outside_distance <= radius,
        'nonfree_rectangles_intersected': collisions, 'nearest_nonfree_rectangle': nearest,
        'scope': 'Exact frozen world center with unchanged robot+arrival+half-cell-diagonal reserve. No motion authority or physical submillimeter accuracy claim.'}


def main():
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument('--run-dir', type=Path, default=ROOT/'outputs/ros_smoke/run31-observation-route')
    parser.add_argument('--closure', type=Path, default=ROOT/'outputs/setup/run31-postclosure-check.json')
    parser.add_argument('--output-dir', type=Path, default=ROOT/'outputs/setup')
    args = parser.parse_args()
    paths = [args.output_dir/('run31-skip-analysis.'+ext) for ext in ('json', 'md')]
    if any(p.exists() or p.is_symlink() for p in paths):
        raise RuntimeError('Refuse overwriting analysis')
    sources = {}
    def read(path):
        raw = path.read_bytes()
        sources[str(path)] = hashlib.sha256(raw).hexdigest()
        return json.loads(raw)
    run = args.run_dir
    mission = read(run/'mission/mission_20261103_014821.json')
    prep, params = read(run/'preparation.json'), read(run/'parameters-early.json')
    closure = read(args.closure)
    assert closure['root_exec_completion']['session_id'] == 26612
    assert closure['root_exec_completion']['exit_code'] == 1
    assert all(g['remaining'] == [] and g['errors'] == [] and g['remaining_check_complete']
               for g in closure['groups'])
    source_matches = {}
    for name in ('src/limo_maze_mission/limo_maze_mission/observation.py',
                 'src/limo_maze_mission/limo_maze_mission/core.py'):
        checksum = hashlib.sha256((ROOT/name).read_bytes()).hexdigest()
        sources[str(ROOT/name)] = checksum
        source_matches[name] = checksum == prep['prepared_source_sha256'][name]
    assert all(source_matches.values()), 'Production source differs from run31'
    values = params['parameters']['limo_maze_mission']['values']
    policy = Policy(*[values[k]['value'] for k in ('robot_radius_m', 'goal_arrival_margin_m',
        'observation_station_spacing_m', 'observation_max_stations', 'observation_headings_per_station',
        'observation_max_dispatches')], max_grid_cells=values['observation_max_grid_cells']['value'])
    snaps = {name: read(run/'observer-snapshots'/name) for name in NAMES}
    initial = snaps[NAMES[0]]
    first_grid = grid(initial)
    queue = ObservationQueue(first_grid, pose(initial), mission['map_epoch'], policy)
    initial_order, actual, matches = list(queue.pending), mission['observation_actions']['attempts'], []
    for attempt in actual:
        view = queue.views[attempt['action_id']]
        matched = (view.station_id == attempt['station_id'] and view.heading_index == attempt['heading_index']
                   and json.loads(json.dumps(asdict(view.goal))) == attempt['goal'])
        matches.append({'action_id': attempt['action_id'], 'all_view_fields_match': matched})
        assert matched
        # The original safe map constructs queue state from actual outcomes;
        # it is NOT claimed as the actual map used at each historical dispatch.
        decision = queue.take_next(first_grid, pose(initial), mission['map_epoch'])
        assert decision.state == 'dispatch' and decision.view.action_id == attempt['action_id']
        assert decision.dispatch_id == attempt['dispatch_id']
        queue.finish(attempt['action_id'], mission['map_epoch'], attempt['outcome'], attempt['dispatch_id'])
    remaining = list(queue.pending)
    metrics = []
    for name, snapshot in snaps.items():
        g = grid(snapshot)
        distances = reachable(g, pose(snapshot), policy)
        radius = endpoint_radius(g, policy)
        base_safe = g.safe_cells(radius)
        stations = []
        for station, xy in enumerate(queue.stations):
            cell = g.to_cell(*xy)
            offset = math.ceil(math.dist(g.to_world(cell), xy)*1e9)/1e9
            effective = endpoint_radius(g, policy, offset)
            endpoint_safe = g.safe_cells(effective)
            stations.append({'station_id': station, 'world_goal_xy': xy, 'goal_cell': cell,
                'goal_cell_value': g.value(cell), 'goal_cell_center_world': g.to_world(cell),
                'offset_ceil_m': offset, 'robot_radius_component_reachable': cell in distances,
                'base_endpoint_radius_m': radius, 'effective_endpoint_radius_m': effective,
                'base_endpoint_clear_without_offset': cell in base_safe,
                'production_endpoint_clear': cell in endpoint_safe,
                'production_skip_on_this_snapshot': bool(distances) and
                    (cell not in distances or cell not in endpoint_safe),
                'exact_world_center_diagnostic': exact_world_endpoint(g, xy, radius),
                'base_endpoint_blockers': blockers(g, cell, radius, first_grid),
                'effective_endpoint_blockers': blockers(g, cell, effective, first_grid)})
        metrics.append({'file': name, 'observer_wall_elapsed_s': snapshot['wall_s'],
            'observer_ros_s': snapshot['ros_s'], 'map': snapshot['map'], 'pose': snapshot['pose'],
            'map_data_sha256': hashlib.sha256(json.dumps(snapshot['map_data'], separators=(',', ':')).encode()).hexdigest(),
            'map_age_ros_s': snapshot['map_age_s'], 'pose_age_ros_s': snapshot['pose_age_s'],
            'reachable_cells': len(distances), 'stations': stations,
            'queue_status_received': snapshot['mission_status']['observation_actions']['queue']})
    replay = deepcopy(queue)
    decision = replay.take_next(grid(snaps[NAMES[4]]), pose(snaps[NAMES[4]]), mission['map_epoch'])
    assert replay.snapshot() == mission['observation_actions']['queue']
    actual_by = {a['action_id']: a for a in actual}
    views = [{**asdict(view), 'reconstructed_final_state': replay.records[action_id]['state'],
        'actual_dispatch_id': actual_by.get(action_id, {}).get('dispatch_id'),
        'actual_outcome': actual_by.get(action_id, {}).get('outcome'),
        'provenance': 'Actual dispatched view matched exactly' if action_id in actual_by else
            'Undispatched view reconstructed from initial observer snapshot; not independently persisted by MissionNode'}
        for action_id, view in replay.views.items()]
    at, last = metrics[4], metrics[-1]
    result = {'schema_version': 1, 'run': 'run31-observation-route', 'runtime_head': prep['code_head'],
        'read_only': True, 'script_sha256': hashlib.sha256(Path(__file__).read_bytes()).hexdigest(),
        'sources': sources, 'source_sha_matches_runtime_preparation': source_matches,
        'inputs_read_scope': 'Six small observer snapshots and mission/parameter/preparation/closure receipts; no observer.json, costmaps, bag, ROS or simulation.',
        'closure': {'root_exec_session': 26612, 'root_exit_code': 1, 'later_groups_all_empty': True,
            'original_all_owned_groups_empty': closure['original_all_owned_groups_empty'], 'original_evidence_modified': False},
        'policy': asdict(policy), 'denominator': 12, 'completed': 5, 'skipped': 7,
        'observed_final_queue': mission['observation_actions']['queue'],
        'reconstruction': {'initial_snapshot': NAMES[0], 'actual_view_matches': matches,
            'stations': queue.stations, 'initial_execution_order': initial_order,
            'pending_after_five_actual_successes': remaining,
            'history_construction_scope': 'Production take_next/finish on the initial safe snapshot with actual outcomes constructs queue state only; historical per-dispatch map identity is not claimed.',
            'final_snapshot': NAMES[4], 'production_decision': asdict(decision),
            'final_queue_matches_recorded': True, 'views': views},
        'findings': {'snapshot_supported_branch': 'endpoint_clearance; robot-radius connectivity remains true',
            'origin_delta_xy_yaw': [at['map']['origin'][i]-initial['map']['origin'][i] for i in range(3)],
            'goal_cell_indices_changed': False, 'offset_alone_explains_failure': False,
            'exact_world_center_same_reserve_stations_1_2_clear': [
                at['stations'][i]['exact_world_center_diagnostic']['circle_clear'] for i in (1, 2)],
            'observed_revalidation_ros_stamp': 278.805, 'decision_time_map_and_pose_not_persisted': True,
            'station_1_safe_again_at_final_snapshot': last['stations'][1]['production_endpoint_clear'],
            'station_2_safe_at_final_snapshot': last['stations'][2]['production_endpoint_clear']},
        'snapshots': metrics,
        'limitations': ['Observer snapshots are independently received, not atomic MissionNode decision inputs.',
            'All five dispatched views match exactly; never-dispatched station1 geometry is reconstructed.',
            'Changed map occupancy does not establish physical obstacle motion versus SLAM/raster changes.',
            'Base endpoint reserve already fails without the additional origin-offset reserve on late snapshots.',
            'All 12 views remain in the denominator. Five completions do not establish visual coverage or mission success.'],
        'next_evidence': 'Bounded per-skip action/station IDs, map stamp/hash/geometry, current pose, goal cell, reachable and clearance booleans, effective radius. Preserve safety thresholds and finite budget.'}
    markdown = """run31 观察跳过分析（只读）

结论：小型 observer 快照能复现最终 **5/12 完成、7/12 跳过**；复算失败分支是**终点净空**，连通性仍通过。快照不是 MissionNode 决策时刻的原子输入，因此不能声称精确故障时刻归因。

使用运行版本 acf3419cfaa78ae994b704f009c3d9cb6d337609 的原 observation.py/core.py，源 SHA 与 preparation 一致。0032 快照重建的前五个实际 action ID、站点、朝向和完整 goal 字段逐值吻合。以实际五次成功结果构造剩余队列，再用0058快照调用原 take_next，得到 exhausted，最终队列与原记录完全一致。

| 站点 | 冻结 world 坐标 | 完成 / 计划 | 剩余跳过 |
| --- | --- | --- | --- |
| 0 | (1.630672, -1.494262) | 4 / 4 | 0 |
| 1 | (3.180672, -0.044262) | 0 / 4 | 4 |
| 2 | (0.330672, 0.005738) | 1 / 4 | 3 |

站1没有实际派发，其几何来自0032快照的生产算法重建，不能称为独立保存的原MissionNode view。

- 0053（ROS245.3、map245.125）三个站点均可达且终点净空通过。
- 0054、0057、0058（ROS276.25、277.1、278.9）三个站点仍在robot_radius=0.25m的安全连通分量内；站0净空通过，站1/2净空失败。
- 基础终点半径0.335355339586m = robot0.25 + arrival0.05 + 半格对角线0.03535534。冻结world目标到当前格心偏移向上取整0.007539168m，实际检查半径0.342894507586m。
- 站1目标格(73,41)附近占用格(75,48)=100、站2目标格(16,42)附近占用格(14,35)=100；两格到各目标格心的矩形最近距离都是0.333541606573m，已进入**基础**终点半径。因此不能把7项skip单独归因于额外offset reserve。
- origin.x从-0.494328080073变为-0.501867247519，变化-0.007539167445m；origin.y/yaw不变，84x52扩展为84x53，目标格索引不变。上述相同索引初始值0、晚期值100，但原点变动意味着相同索引不是完全相同world矩形，不能由此宣称真实障碍移动。

下一步优先保留有界逐项skip诊断：action/station、map stamp/hash/geometry、当前pose、goal cell、reachable、effective endpoint radius、clearance boolean。**不建议猜测性缩小安全边界。**

同名JSON包含12项清单、逐快照判据、阻挡格与输入SHA。analyze-run31-skips.py支持--run-dir、--closure和--output-dir，以便使用保留相对布局的compact archive复算；输出禁止覆盖，源码必须匹配原运行SHA。
"""
    for path, value in zip(paths, (json.dumps(result, ensure_ascii=False, indent=2, allow_nan=False)+'\n', markdown)):
        with path.open('x', encoding='utf-8', newline='\n') as stream:
            stream.write(value)
    print(json.dumps({'written': [str(p) for p in paths], 'snapshots': list(NAMES),
        'source_sha_matches': source_matches, 'production_replay_matches_final': True,
        'completed': 5, 'skipped': 7, 'denominator': 12, 'bag_decoded': False}, indent=2))


if __name__ == '__main__':
    main()
