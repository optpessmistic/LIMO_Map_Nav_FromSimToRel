import hashlib
import importlib
import json
from pathlib import Path

root = Path.cwd()
names = ['limo_evidence_protocol.retention', 'limo_maze_mission.retention', 'limo_maze_mission.node',
         'limo_maze_perception.node', 'limo_maze_perception.retention_adapter', 'limo_maze_perception.retention_worker']
names += ['limo_maze_mission.core','limo_maze_mission.inspection','limo_maze_mission.observation','limo_maze_perception.evidence']
rows = []
for name in names:
    module = importlib.import_module(name)
    package, filename = name.split('.')
    installed = Path(module.__file__).resolve()
    source = root/'src'/package/package/(filename+'.py')
    rows.append(dict(module=name, installed=str(installed), source=str(source),
                     sha256=hashlib.sha256(source.read_bytes()).hexdigest(),
                     equal=installed.read_bytes() == source.read_bytes(),
                     current_workspace=installed.is_relative_to(root/'install')))
from ament_index_python.packages import get_package_share_directory
launch_rows=[]
for package,filename in (('limo_maze_mission','mission.launch.py'),
        ('limo_maze_bringup','mapping_navigation.launch.py'),
        ('limo_maze_bringup','navigation.launch.py'),('limo_sim_bringup','sim.launch.py')):
    source=root/'src'/package/'launch'/filename
    installed=Path(get_package_share_directory(package))/'launch'/filename
    launch_rows.append(dict(package=package,file=filename,source=str(source),installed=str(installed),
        sha256=hashlib.sha256(source.read_bytes()).hexdigest(),
        equal=installed.read_bytes()==source.read_bytes(),
        current_workspace=installed.resolve().is_relative_to(root/'install')))
passed = all(r['equal'] and r['current_workspace'] for r in rows+launch_rows)
report = dict(passed=passed, modules=rows, launch_files=launch_rows)

from ament_index_python.packages import get_package_prefix
binding = json.loads((root/'outputs/ros_smoke/run30-steady-tf/preparation.json').read_text())['slam_overlay']
slam_files = {name: hashlib.sha256((root/name).read_bytes()).hexdigest()
              for name in binding['files_sha256']}
slam_passed = (get_package_prefix('slam_toolbox') == binding['package_prefix']
    and slam_files == binding['files_sha256'])
passed = passed and slam_passed
report.update(passed=passed, slam_overlay={'passed':slam_passed,
    'resolved_package_prefix':get_package_prefix('slam_toolbox'),
    'file_sha256':slam_files, 'scope':'Disk and package resolution only; live /proc/maps checked separately.'})
with (root/'outputs/ros_smoke/run30-steady-tf/installed-check.json').open('x') as out:
    json.dump(report, out, indent=2)
print(json.dumps(report, indent=2))
raise SystemExit(0 if passed else 1)
