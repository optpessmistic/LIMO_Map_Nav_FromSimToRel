SOURCES = {}


def sha(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()


def source(path):
    path = Path(path)
    SOURCES[str(path.relative_to(REPO))] = {'bytes': path.stat().st_size, 'sha256': sha(path)}
    return path


def load(name):
    return json.loads(source(RUN/name).read_text())


def delta(a, b):
    return math.atan2(math.sin(a-b), math.cos(a-b))


def native_delta(a, b):
    return {'dx_m': b['x']-a['x'], 'dy_m': b['y']-a['y'],
            'xy_distance_m': math.hypot(b['x']-a['x'], b['y']-a['y']),
            'yaw_delta_rad': delta(b['yaw'], a['yaw'])}


def span(samples):
    return {'count': len(samples), 'first_sim_s': samples[0]['sim_s'],
            'last_sim_s': samples[-1]['sim_s'],
            'max_gap_s': max((b['sim_s']-a['sim_s'] for a,b in zip(samples,samples[1:])), default=0),
            'x_span_m': max(s['x'] for s in samples)-min(s['x'] for s in samples),
            'y_span_m': max(s['y'] for s in samples)-min(s['y'] for s in samples),
            'yaw_span_rad': max(delta(s['yaw'],samples[0]['yaw']) for s in samples)-
                            min(delta(s['yaw'],samples[0]['yaw']) for s in samples)}
